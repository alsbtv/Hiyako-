package com.hiyako.app

import android.app.NotificationChannel
import android.app.NotificationManager
import android.app.Service
import android.content.Intent
import android.os.Build
import android.os.IBinder
import android.os.PowerManager
import androidx.core.app.NotificationCompat

class NexusService : Service() {
    private var wakeLock: PowerManager.WakeLock? = null

    override fun onBind(intent: Intent?): IBinder? = null

    override fun onCreate() {
        super.onCreate()
        val powerManager = getSystemService(POWER_SERVICE) as PowerManager
        wakeLock = powerManager.newWakeLock(PowerManager.PARTIAL_WAKE_LOCK, "NexusAI::ProcessingWakeLock")
        wakeLock?.acquire(10 * 60 * 1000L) // 10 minutes max
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        val channelId = "nexus_fg_channel"
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val channel = NotificationChannel(channelId, "Nexus AI Processing", NotificationManager.IMPORTANCE_LOW)
            getSystemService(NotificationManager::class.java).createNotificationChannel(channel)
        }

        val notification = NotificationCompat.Builder(this, channelId)
            .setContentTitle("Nexus AI is working...")
            .setContentText("Generating response in the background")
            .setSmallIcon(android.R.drawable.ic_dialog_info)
            .setOngoing(true)
            .build()

        startForeground(1, notification)
        return START_NOT_STICKY
    }

    override fun onDestroy() {
        wakeLock?.let { if (it.isHeld) it.release() }
        super.onDestroy()
    }
}

interface PuterListener {

    fun onModelsReceived(json: String)
    fun onChunk(reqId: String, text: String)
    fun onDone(reqId: String)
    fun onError(reqId: String, message: String)

/**
 * Owns the hidden WebView that runs Puter.js. It lives as long as the ViewModel,
 * so switching screens never destroys the page or an in-flight request.
 */
class PuterEngine(context: Context, private val listener: PuterListener) {

    private val appContext: Context = context.applicationContext

    private val _ready = MutableStateFlow(false)
    val ready: StateFlow<Boolean> = _ready.asStateFlow()

    private val _popup = MutableStateFlow<WebView?>(null)
    val popup: StateFlow<WebView?> = _popup.asStateFlow()

    val webView: WebView

    init {
        webView = createWebView()
    }

    @SuppressLint("SetJavaScriptEnabled")
    private fun createWebView(): WebView {
        val wv = WebView(appContext)
        wv.setBackgroundColor(android.graphics.Color.TRANSPARENT)
        wv.settings.javaScriptEnabled = true
        wv.settings.domStorageEnabled = true
        wv.settings.setSupportMultipleWindows(true)
        wv.settings.javaScriptCanOpenWindowsAutomatically = true
        wv.webViewClient = WebViewClient()
        wv.webChromeClient = object : WebChromeClient() {
            override fun onCreateWindow(
                view: WebView?,
                isDialog: Boolean,
                isUserGesture: Boolean,
                resultMsg: Message?
            ): Boolean {
                if (resultMsg == null) return false
                val popupView = WebView(appContext)
                popupView.settings.javaScriptEnabled = true
                popupView.settings.domStorageEnabled = true
                popupView.settings.javaScriptCanOpenWindowsAutomatically = true
                popupView.webViewClient = WebViewClient()
                popupView.webChromeClient = object : WebChromeClient() {
                    override fun onCloseWindow(window: WebView?) {
                        closePopup()
                    }
                }
                val transport = resultMsg.obj as WebView.WebViewTransport
                transport.webView = popupView
                resultMsg.sendToTarget()
                _popup.value = popupView
                return true
            }
        }
        wv.addJavascriptInterface(Bridge(), "AndroidBridge")
        wv.loadDataWithBaseURL(BASE_URL, PAGE_HTML, "text/html", "UTF-8", null)
        return wv
    }

    fun closePopup() {
        val p = _popup.value
        _popup.value = null
        if (p != null) {
            p.post {
                try {
                    p.stopLoading()
                    p.loadUrl("about:blank")
                } catch (e: Exception) {
                    // ignore
                }
            }
        }
    }

    fun ask(reqId: String, messagesJson: String, model: String, hasImage: Boolean) {
        val js = "window.ask(" + JSONObject.quote(reqId) + "," + JSONObject.quote(messagesJson) + "," +
            JSONObject.quote(model) + "," + hasImage + ");"
        run(js)
    }

    fun makeTitle(convoId: Int, text: String) {
        run("window.makeTitle(" + convoId + "," + JSONObject.quote(text) + ");")
    }

    fun refreshModels() {
        run("window.fetchModels();")
    }

    fun checkAuth() {
        run("window.checkAuth();")
    }

    fun signIn() {
        run("window.signIn();")
    }

    fun reload() {
        _ready.value = false
        webView.post { webView.loadDataWithBaseURL(BASE_URL, PAGE_HTML, "text/html", "UTF-8", null) }
    }

    fun destroy() {
        try {
            webView.destroy()
        } catch (e: Exception) {
            // ignore
        }
    }

    private fun run(js: String) {
        webView.post { webView.evaluateJavascript(js, null) }
    }

    inner class Bridge {
        @JavascriptInterface
        fun onReady() {
            _ready.value = true
            listener.onEngineReady()
        }

        @JavascriptInterface
        fun onModels(json: String) {
            listener.onModelsReceived(json)
        }

        @JavascriptInterface
        fun onChunk(reqId: String, text: String) {
            listener.onChunk(reqId, text)
        }

        @JavascriptInterface
        fun onDone(reqId: String) {
            listener.onDone(reqId)
        }

        @JavascriptInterface
        fun onError(reqId: String, message: String) {
            listener.onError(reqId, message)
        }

        @JavascriptInterface
        fun onTitle(convoId: String, title: String) {
            val id = convoId.toIntOrNull() ?: return
            listener.onTitle(id, title)
        }

        @JavascriptInterface
        fun onAuth(signedIn: String, username: String) {
            listener.onAuth(signedIn == "true", username)
        }

        @JavascriptInterface
        fun getImage(reqId: String): String {
            return listener.provideImage(reqId)
        }
    }

    companion object {
        // Keep this base URL identical to the old version so an existing Puter session in localStorage survives.
        private const val BASE_URL = "https://appassets.androidplatform.net"

        private val PAGE_HTML = """
            <html><head><meta charset='utf-8'><script src='https://js.puter.com/v2/'></script></head><body>
            <script>
            function errText(e) {
                try {
                    if (!e) return 'Unknown error';
                    if (typeof e === 'string') return e;
                    if (e.message) return e.message;
                    if (e.error && e.error.message) return e.error.message;
                    if (e.error && typeof e.error === 'string') return e.error;
                    return JSON.stringify(e);
                } catch (x) { return 'Unknown error'; }
            }
            function contentText(c) {
                if (c === null || c === undefined) return '';
                if (typeof c === 'string') return c;
                if (Array.isArray(c)) {
                    return c.map(function (p) { return (p && p.text) ? p.text : (typeof p === 'string' ? p : ''); }).join('');
                }
                if (c.text) return c.text;
                return JSON.stringify(c);
            }
            function respText(r) {
                if (r === null || r === undefined) return '';
                if (typeof r === 'string') return r;
                if (r.message) return contentText(r.message.content);
                if (r.text) return r.text;
                return JSON.stringify(r);
            }
            window.fetchModels = function () {
                try {
                    puter.ai.listModels().then(function (models) {
                        var result = models.map(function (m) {
                            if (typeof m === 'string') return { id: m, name: m };
                            return { id: m.id, name: m.name || m.id };
                        });
                        AndroidBridge.onModels(JSON.stringify(result));
                    }).catch(function (e) {});
                } catch (e) {}
            };
            window.ask = async function (reqId, msgsJson, model, hasImage) {
                try {
                    var msgs = JSON.parse(msgsJson);
                    var opts = {};
                    if (model) { opts.model = model; }
                    var resp;
                    if (hasImage) {
                        var img = AndroidBridge.getImage(reqId);
                        var prompt = contentText(msgs[msgs.length - 1].content);
                        resp = await puter.ai.chat(prompt, img, false, opts);
                    } else {
                        opts.stream = true;
                        resp = await puter.ai.chat(msgs, false, opts);
                    }
                    if (resp && resp.success === false) { throw resp; }
                    if (resp && typeof resp[Symbol.asyncIterator] === 'function') {
                        for await (var part of resp) {
                            if (part && part.text) { AndroidBridge.onChunk(reqId, part.text); }
                        }
                    } else {
                        var t = respText(resp);
                        if (t) { AndroidBridge.onChunk(reqId, t); }
                    }
                    AndroidBridge.onDone(reqId);
                } catch (e) {
                    AndroidBridge.onError(reqId, errText(e));
                }
            };
            window.makeTitle = function (convoId, text) {
                try {
                    var p = 'Write a short title (2 to 5 words) for a conversation that starts with the message below. Use the same language as the message. Output ONLY the title, with no quotes and no trailing punctuation.\n\nMessage: ' + String(text).substring(0, 300);
                    puter.ai.chat(p).then(function (r) {
                        var t = respText(r);
                        if (t) { AndroidBridge.onTitle(String(convoId), t.trim()); }
                    }).catch(function (e) {});
                } catch (e) {}
            };
            window.checkAuth = function () {
                try {
                    if (puter.auth.isSignedIn()) {
                        puter.auth.getUser().then(function (u) {
                            AndroidBridge.onAuth('true', (u && u.username) ? u.username : '');
                        }).catch(function (e) { AndroidBridge.onAuth('true', ''); });
                    } else {
                        AndroidBridge.onAuth('false', '');
                    }
                } catch (e) { AndroidBridge.onAuth('false', ''); }
            };
            window.signIn = function () {
                try {
                    puter.auth.signIn().then(function () {
                        window.checkAuth();
                        window.fetchModels();
                    }).catch(function (e) {});
                } catch (e) {}
            };
            (function boot() {
                if (typeof puter === 'undefined') { setTimeout(boot, 300); return; }
                AndroidBridge.onReady();
                window.fetchModels();
                window.checkAuth();
            })();
            </script></body></html>
        """.trimIndent()
    }
}

// ---------------------------------------------------------------------------
// Lightweight Markdown rendering (headings, lists, bold, inline code, code fences)
// ---------------------------------------------------------------------------

sealed interface MdBlock
data class MdText(val text: String, val heading: Int = 0, val marker: String? = null) : MdBlock
data class MdCode(val lang: String, val code: String) : MdBlock

private val numberedRegex = Regex("^(\\d+)[.)]\\s+(.*)$")

fun parseMarkdown(src: String): List<MdBlock> {
    val out = mutableListOf<MdBlock>()
    val lines = src.lines()
    var i = 0
    while (i < lines.size) {
        val line = lines[i]
        val trimmed = line.trimStart()
        if (trimmed.startsWith("```")) {
            val lang = trimmed.removePrefix("```").trim()
            val sb = StringBuilder()
            i++
            while (i < lines.size && !lines[i].trimStart().startsWith("```")) {
                if (sb.isNotEmpty()) sb.append('\n')
                sb.append(lines[i])
                i++
            }
            out.add(MdCode(if (lang.isBlank()) "code" else lang, sb.toString()))
            i++
            continue
        }
        when {
            line.isBlank() -> {}
            trimmed.startsWith("### ") -> out.add(MdText(trimmed.removePrefix("### "), 3))
            trimmed.startsWith("## ") -> out.add(MdText(trimmed.removePrefix("## "), 2))
            trimmed.startsWith("# ") -> out.add(MdText(trimmed.removePrefix("# "), 1))
            trimmed.startsWith("- ") || trimmed.startsWith("* ") -> out.add(MdText(trimmed.substring(2), 0, "•"))
            else -> {
                val m = numberedRegex.matchEntire(trimmed)
                if (m != null) {
                    out.add(MdText(m.groupValues[2], 0, m.groupValues[1] + "."))
                } else {
                    out.add(MdText(line, 0))
                }
            }
        }
        i++
    }
    return out
}

private fun inlineMarkdown(src: String, codeBackground: Color, codeColor: Color): AnnotatedString {
    return buildAnnotatedString {
        var i = 0
        while (i < src.length) {
            if (src.startsWith("**", i)) {
                val end = src.indexOf("**", i + 2)
                if (end > i + 2) {
                    withStyle(SpanStyle(fontWeight = FontWeight.Bold)) { append(src.substring(i + 2, end)) }
                    i = end + 2
                    continue
                }
            }
            if (src[i] == '`') {
                val end = src.indexOf('`', i + 1)
                if (end > i + 1) {
                    withStyle(SpanStyle(fontFamily = FontFamily.Monospace, background = codeBackground, color = codeColor)) {
                        append(src.substring(i + 1, end))
                    }
                    i = end + 1
                    continue
                }
            }
            append(src[i])
            i++
        }
    }
}

@Composable
fun MarkdownText(text: String) {
    val blocks = remember(text) { parseMarkdown(text) }
    Column(Modifier.fillMaxWidth()) {
        blocks.forEach { block ->
            when (block) {
                is MdCode -> CollapsibleCodeBlock(block.code, block.lang)
                is MdText -> {
                    val styled = inlineMarkdown(block.text, ElevatedSurface, TextPrimary)
                    val size = when (block.heading) {
                        1 -> 21.sp
                        2 -> 18.sp
                        3 -> 16.sp
                        else -> 15.sp
                    }
                    val weight = if (block.heading > 0) FontWeight.Bold else FontWeight.Normal
                    val marker = block.marker
                    if (marker != null) {
                        Row(Modifier.fillMaxWidth().padding(vertical = 2.dp)) {
                            Text(marker, color = TextSecondary, fontSize = size, lineHeight = 24.sp, modifier = Modifier.width(26.dp))
                            Text(
                                styled,
                                color = TextPrimary,
                                fontSize = size,
                                lineHeight = 24.sp,
                                fontWeight = weight,
                                modifier = Modifier.weight(1f)
                            )
                        }
                    } else {
                        Text(
                            styled,
                            color = TextPrimary,
                            fontSize = size,
                            lineHeight = 24.sp,
                            fontWeight = weight,
                            modifier = Modifier.fillMaxWidth().padding(vertical = 2.dp)
                        )
                    }
                }
            }
        }
    }
}

    fun loadToken(context: Context) {
        val prefs = context.getSharedPreferences("nexus_prefs", Context.MODE_PRIVATE)
        _token.value = prefs.getString("github_token", "") ?: ""
        if (_token.value.isNotBlank()) fetchRepos()
    }

    fun saveToken(context: Context, token: String) {
        val prefs = context.getSharedPreferences("nexus_prefs", Context.MODE_PRIVATE)
        prefs.edit().putString("github_token", token).apply()
        _token.value = token
        fetchRepos()
    }

    fun fetchRepos() {
        if (_token.value.isBlank()) return
        _isLoading.value = true
        viewModelScope.launch {
            _repos.value = GitHubApi.getUserRepos(_token.value)
            _isLoading.value = false
        }
    }

    fun selectRepoAndLoadFiles(repo: Repo, onComplete: () -> Unit) {
        _isLoading.value = true
        _loadingMessage.value = "Fetching file tree..."
        viewModelScope.launch {
            val owner = repo.fullName.split("/")[0]
            val filesList = GitHubApi.getRepoFiles(_token.value, owner, repo.name, repo.defaultBranch)
            val loadedFiles = mutableMapOf<String, String>()
            
            // تم إزالة الحد الأقصى لقراءة كل الملفات
            filesList.forEachIndexed { index, path -> 
                _loadingMessage.value = "Loading files: ${index + 1}/${filesList.size}..."
                val content = GitHubApi.getFileContent(_token.value, owner, repo.name, path)
                if (content != null) {
                    loadedFiles[path] = content
                }
            }
            
            WorkspaceManager.activeRepo = repo.fullName
            WorkspaceManager.projectFiles = loadedFiles
            WorkspaceManager.isLoaded = loadedFiles.isNotEmpty()
            
            _isLoading.value = false
            _loadingMessage.value = ""
            onComplete()
        }
    }

@Composable
private fun ModelPickerSheet(
    models: List<AiModel>,
    selectedId: String,
    onSelect: (AiModel) -> Unit,
    onRefresh: () -> Unit,
    onDismiss: () -> Unit
) {
    val sheetState = rememberModalBottomSheetState(skipPartiallyExpanded = true)

        if (q.isBlank()) models else models.filter { it.name.contains(q, true) || it.id.contains(q, true) }
    }
    ModalBottomSheet(onDismissRequest = onDismiss, sheetState = sheetState, containerColor = CardDark) {
        Column(Modifier.fillMaxWidth().fillMaxHeight(0.8f).padding(horizontal = 16.dp)) {

                Text(
                    "No models loaded yet. Check your internet connection and Puter sign-in, then refresh.",
                    color = TextSecondary,
                    fontSize = 14.sp

                TextButton(onClick = onRefresh) { Text("Refresh", color = AccentBlue) }

                SearchField(q, { q = it }, "Search models...")
                Spacer(Modifier.height(8.dp))
                LazyColumn(Modifier.weight(1f)) {

            .trim()

    fun sendMessage(text: String, attachmentUri: Uri?) {
        val convoId = _activeConversationId.value ?: return
        val trimmed = text.trim()
        if (trimmed.isEmpty() && attachmentUri == null) return
        if (!markGenerating(convoId)) return

        val currentTitle = _conversations.value.firstOrNull { it.id == convoId }?.title
        val needsTitle = currentTitle == null || currentTitle == DEFAULT_TITLE || currentTitle == "New Group"

        viewModelScope.launch {
            try {
                var type = "TEXT"
                var storedUri: String? = null
                var displayName: String? = null
                if (attachmentUri != null) {
                    val saved = withContext(Dispatchers.IO) { copyAttachment(attachmentUri) }
                    type = if (saved.mime.startsWith("image/")) "IMAGE" else "FILE"
                    storedUri = Uri.fromFile(saved.file).toString()
                    displayName = saved.name
                }
                chatDao.insertMessage(
                    ChatEntity(
                        conversationId = convoId,
                        role = "user",
                        content = trimmed,
                        messageType = type,
                        attachmentUri = storedUri,
                        senderName = displayName
                    )
                )
                conversationDao.touch(convoId, System.currentTimeMillis())

                if (needsTitle) {
                    val seed = if (trimmed.isNotEmpty()) trimmed else (displayName ?: "Attachment")
                    val fallback = seed.lines().first().trim().take(32).ifBlank { DEFAULT_TITLE }
                    titleFallbacks[convoId] = fallback
                    conversationDao.updateTitle(convoId, fallback)
                    engine.makeTitle(convoId, seed)
                }
            } catch (e: Exception) {
                _generating.update { it - convoId }
                _errors.update { it + (convoId to ("Could not attach the file: " + (e.message ?: "unknown error"))) }
                return@launch
            }
            runRequest(convoId)
        }
    }

    fun regenerate() {
        val convoId = _activeConversationId.value ?: return
        if (!markGenerating(convoId)) return
        viewModelScope.launch {
            try {
                val last = chatDao.getMessagesOnce(convoId).lastOrNull()
                if (last != null && last.role != "user") chatDao.deleteMessage(last.id)
            } catch (e: Exception) {
                _generating.update { it - convoId }
                return@launch
            }
            runRequest(convoId)
        }
    }

    fun stopGeneration() {
        val convoId = _activeConversationId.value ?: return
        val partials = mutableListOf<String>()
        pending.entries.toList().forEach { entry ->
            if (entry.value.convoId == convoId) {
                pending.remove(entry.key)
                pendingImages.remove(entry.key)
                partials.add(entry.value.text())
            }
        }
        val partial = partials.joinToString("").trim()
        viewModelScope.launch {
            if (partial.isNotEmpty()) {
                chatDao.insertMessage(ChatEntity(conversationId = convoId, role = "ai", content = partial))
            }
            _generating.update { it - convoId }
            stopServiceIfIdle()
        }
    }

    private suspend fun runRequest(convoId: Int) {
        requestCounter += 1
        val reqId = "r$requestCounter"
        pending[reqId] = Pending(convoId)
        startServiceSafely()
        try {
            val entities = chatDao.getMessagesOnce(convoId)
            val built = withContext(Dispatchers.IO) { buildRequest(entities) }
            if (!pending.containsKey(reqId)) return
            val image = built.imageDataUrl
            if (image != null) pendingImages[reqId] = image
            engine.ask(reqId, built.messagesJson, _selectedModel.value.id, image != null)
        } catch (e: Exception) {
            failRequest(reqId, e.message ?: "Unexpected error")
        }
    }

    private fun buildRequest(entities: List<ChatEntity>): BuiltRequest {
        val system = StringBuilder()
        system.append("You are Nexus AI, a helpful assistant. Reply in the same language as the user's last message. ")
        system.append("Use Markdown for formatting and fenced code blocks for code.")

        val mem = _memories.value.take(20)
        if (mem.isNotEmpty()) {
            system.append("\n\nKnown facts about the user:\n")
            mem.forEach { system.append("- ").append(it.content.take(300)).append('\n') }
        }
        if (isThinkingMode) {
            system.append("\n\nThink through the problem carefully and step by step before giving the final answer.")
        }
        if (isSearchMode) {
            system.append("\n\nThe user turned on Search mode, but you cannot browse the web in this app. ")
            system.append("If the answer depends on recent information, say so clearly instead of guessing.")
        }

        val arr = JSONArray()
        arr.put(JSONObject().put("role", "system").put("content", system.toString()))

        val recent = entities
            .filterNot { it.role == "ai" && it.content == "Thinking..." }
            .takeLast(MAX_HISTORY)

        for (e in recent) {
            val role = if (e.role == "user") "user" else "assistant"
            var content = e.content.take(12000)
            if (role == "user" && e.messageType == "FILE") {
                val name = e.senderName ?: "file"
                val fileText = readTextAttachment(e.attachmentUri)
                content += if (fileText != null) {
                    "\n\n[Attached file: $name]\n```\n$fileText\n```"
                } else {
                    "\n\n[Attached file: $name - this file type cannot be read in this version of the app]"
                }
            }
            arr.put(JSONObject().put("role", role).put("content", content))
        }

        var imageDataUrl: String? = null
        val last = recent.lastOrNull()
        if (last != null && last.role == "user" && last.messageType == "IMAGE") {
            val path = last.attachmentUri?.let { Uri.parse(it).path }
            if (path != null) imageDataUrl = imageToDataUrl(File(path))
            if (imageDataUrl != null && last.content.isBlank()) {
                arr.remove(arr.length() - 1)
                arr.put(JSONObject().put("role", "user").put("content", "Describe this image in detail."))
            }
        }
        return BuiltRequest(arr.toString(), imageDataUrl)
    }

    private fun copyAttachment(uri: Uri): SavedAttachment {
        val name = queryDisplayName(appCtx, uri)
        val mime = appCtx.contentResolver.getType(uri) ?: "application/octet-stream"
        val dir = File(appCtx.filesDir, "attachments")
        dir.mkdirs()
        val safe = name.replace(Regex("[^A-Za-z0-9._-]"), "_")
        val out = File(dir, System.currentTimeMillis().toString() + "_" + safe)
        val input = appCtx.contentResolver.openInputStream(uri) ?: throw IllegalStateException("Cannot open the selected file")
        input.use { src -> out.outputStream().use { dst -> src.copyTo(dst) } }
        return SavedAttachment(out, mime, name)
    }

    private fun readTextAttachment(uriString: String?): String? {
        if (uriString == null) return null
        val path = Uri.parse(uriString).path ?: return null
        val file = File(path)
        if (!file.exists() || file.length() > 5_000_000L) return null
        val bytes = file.readBytes()
        for (b in bytes) {
            if (b.toInt() == 0) return null
        }
        return String(bytes, Charsets.UTF_8).take(40000)
    }

    private fun imageToDataUrl(file: File): String? {
        if (!file.exists()) return null
        val bounds = BitmapFactory.Options()
        bounds.inJustDecodeBounds = true
        BitmapFactory.decodeFile(file.absolutePath, bounds)
        var sample = 1
        while (bounds.outWidth / sample > 1600 || bounds.outHeight / sample > 1600) sample *= 2
        val opts = BitmapFactory.Options()
        opts.inSampleSize = sample
        val bitmap = BitmapFactory.decodeFile(file.absolutePath, opts) ?: return null
        val out = ByteArrayOutputStream()
        bitmap.compress(Bitmap.CompressFormat.JPEG, 85, out)
        bitmap.recycle()
        return "data:image/jpeg;base64," + Base64.encodeToString(out.toByteArray(), Base64.NO_WRAP)
    }

    // ------------------------------------------------------------------ Foreground service

    private fun startServiceSafely() {
        try {
            ContextCompat.startForegroundService(appCtx, Intent(appCtx, NexusService::class.java))
        } catch (e: Exception) {
            // The service only keeps the process alive in the background; chat works without it.
        }
    }

    private fun stopServiceIfIdle() {
        viewModelScope.launch {
            delay(1500)
            if (pending.isEmpty()) {
                try {
                    appCtx.stopService(Intent(appCtx, NexusService::class.java))
                } catch (e: Exception) {
                    // ignore
                }
            }
        }
    }

    private suspend fun conversationExists(id: Int): Boolean = _conversations.value.any { it.id == id }

    private fun failRequest(reqId: String, message: String) {
        val p = pending.remove(reqId) ?: return
        pendingImages.remove(reqId)
        val partial = p.text().trim()
        viewModelScope.launch {
            if (partial.isNotEmpty() && conversationExists(p.convoId)) {
                chatDao.insertMessage(ChatEntity(conversationId = p.convoId, role = "ai", content = partial))
            }
            _errors.update { it + (p.convoId to message) }
            _generating.update { it - p.convoId }
            stopServiceIfIdle()
        }
    }

    // ------------------------------------------------------------------ PuterListener (called from the WebView bridge thread)

    override fun onEngineReady() {}

    override fun onModelsReceived(json: String) {
        try {
            val arr = JSONArray(json)
            val list = mutableListOf<AiModel>()
            for (i in 0 until arr.length()) {
                val o = arr.getJSONObject(i)
                val id = o.optString("id")
                if (id.isNotBlank()) {
                    val name = o.optString("name", id).ifBlank { id }
                    list.add(AiModel(name, id))
                }
            }
            val sorted = list.distinctBy { it.id }.sortedBy { it.name.lowercase() }
            _availableModels.value = listOf(AiModel("Default (Auto)", "")) + sorted
        } catch (e: Exception) {
            // ignore malformed payload
        }
    }

    override fun onChunk(reqId: String, text: String) {
        val p = pending[reqId] ?: return
        val snapshot = p.append(text)
        _generating.update { it + (p.convoId to snapshot) }
    }

    override fun onDone(reqId: String) {
        val p = pending.remove(reqId) ?: return
        pendingImages.remove(reqId)
        val finalText = p.text().trim()
        viewModelScope.launch {
            if (conversationExists(p.convoId)) {
                if (finalText.isNotEmpty()) {
                    chatDao.insertMessage(ChatEntity(conversationId = p.convoId, role = "ai", content = finalText))
                    conversationDao.touch(p.convoId, System.currentTimeMillis())
                } else {
                    _errors.update { it + (p.convoId to "The model returned an empty reply.") }
                }
            }
            _generating.update { it - p.convoId }
            stopServiceIfIdle()
        }
    }

    override fun onError(reqId: String, message: String) {
        failRequest(reqId, message)
    }

    override fun onTitle(convoId: Int, title: String) {
        val clean = title.trim().trim('"', '\'', '«', '»').trim().take(60)
        if (clean.isEmpty()) return
        val fallback = titleFallbacks[convoId] ?: return
        val current = _conversations.value.firstOrNull { it.id == convoId }?.title
        if (current == fallback) {
            titleFallbacks.remove(convoId)
            viewModelScope.launch { conversationDao.updateTitle(convoId, clean) }
        }
    }

    override fun onAuth(signedIn: Boolean, username: String) {
        _auth.value = AuthInfo(signedIn, username)
    }

    override fun provideImage(reqId: String): String = pendingImages[reqId] ?: ""

    override fun onCleared() {
        tts?.shutdown()
        if (engineHolder.isInitialized()) engine.destroy()
        super.onCleared()
    }

    companion object {
        private const val DEFAULT_TITLE = "New Chat"
        private const val REQUEST_TIMEOUT_MS = 120_000L
        private const val MAX_HISTORY = 30
    }

@Composable
private fun AppRoot(vm: AiViewModel) {
    var screenState by remember { mutableStateOf("chat") }
    var showEngineHost by remember { mutableStateOf(false) }

    // Create the hidden Puter WebView only after the first frame so app start-up is not delayed.
    LaunchedEffect(Unit) {
        withFrameNanos { }
        showEngineHost = true
    }

    BackHandler(enabled = screenState != "chat") {
        screenState = if (screenState == "settings") "chat" else "settings"
    }

    Box(Modifier.fillMaxSize()) {
        when (screenState) {
            "workspace" -> WorkspaceScreen(
                onBack = { screenState = "settings" },
                onRepoSelected = { screenState = "chat" },
                onNavigateToUpdater = { screenState = "updater" }
            )
            "updater" -> UpdaterScreen(onBack = { screenState = "settings" })
            "memory" -> MemoryScreen(onBack = { screenState = "settings" }, vm = vm)
            "settings" -> SettingsScreen(
                onBack = { screenState = "chat" },
                onNavigateToWorkspace = { screenState = "workspace" },
                onNavigateToUpdater = { screenState = "updater" },
                onNavigateToMemory = { screenState = "memory" },
                onSignIn = {
                    vm.signIn()
                    screenState = "chat"
                },
                onClearChat = { vm.clearAllChats() },
                vm = vm
            )
            else -> ChatScreen(onNavigateToSettings = { screenState = "settings" }, vm = vm)
        }

        if (showEngineHost) {
            EngineHost(vm)
        }
    }
}

package com.hiyako.app.data

import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import okhttp3.MediaType.Companion.toMediaType
import okhttp3.OkHttpClient
import okhttp3.Request
import okhttp3.RequestBody.Companion.toRequestBody
import org.json.JSONArray
import org.json.JSONObject
import java.util.Base64

data class Repo(val name: String, val fullName: String, val defaultBranch: String)

object GitHubApi {
    private val client = OkHttpClient.Builder().build()
    private val jsonMediaType = "application/json; charset=utf-8".toMediaType()

    private fun authRequest(url: String, token: String) = Request.Builder()
        .url(url)
        .header("Authorization", "Bearer $token")
        .header("Accept", "application/vnd.github.v3+json")
        .build()

    suspend fun getUserRepos(token: String): List<Repo> = withContext(Dispatchers.IO) {
        try {
            client.newCall(authRequest("https://api.github.com/user/repos", token)).execute().use { response ->
                if (!response.isSuccessful) return@withContext emptyList()
                val jsonArray = JSONArray(response.body?.string() ?: "[]")
                val repos = mutableListOf<Repo>()
                for (i in 0 until jsonArray.length()) {
                    val repo = jsonArray.getJSONObject(i)
                    repos.add(Repo(repo.getString("name"), repo.getString("full_name"), repo.getString("default_branch")))
                }
                return@withContext repos
            }
        } catch (e: Exception) { return@withContext emptyList() }
    }

    suspend fun getRepoFiles(token: String, owner: String, repo: String, branch: String): List<String> = withContext(Dispatchers.IO) {
        val files = mutableListOf<String>()
        try {
            val url = "https://api.github.com/repos/$owner/$repo/git/trees/$branch?recursive=1"
            client.newCall(authRequest(url, token)).execute().use { response ->
                if (!response.isSuccessful) return@withContext emptyList()
                val tree = JSONObject(response.body?.string() ?: "{}").optJSONArray("tree") ?: return@withContext emptyList()
                for (i in 0 until tree.length()) {
                    val item = tree.getJSONObject(i)
                    if (item.getString("type") == "blob") {
                        val path = item.getString("path")
                        if (!path.contains("build/") && !path.contains(".git/") && (path.endsWith(".kt") || path.endsWith(".xml") || path.endsWith(".kts") || path.endsWith(".gradle"))) {
                            files.add(path)
                        }
                    }
                }
            }
        } catch (e: Exception) { e.printStackTrace() }
        return@withContext files
    }

    suspend fun getFileContent(token: String, owner: String, repo: String, path: String): String? = withContext(Dispatchers.IO) {
        try {
            val url = "https://api.github.com/repos/$owner/$repo/contents/$path"
            client.newCall(authRequest(url, token)).execute().use { response ->
                if (!response.isSuccessful) return@withContext null
                val contentBase64 = JSONObject(response.body?.string() ?: "{}").optString("content").replace("\n", "")
                if (contentBase64.isBlank()) return@withContext null
                String(Base64.getDecoder().decode(contentBase64))
            }
        } catch (e: Exception) { return@withContext null }
    }

    private suspend fun getFileSha(token: String, owner: String, repo: String, path: String): String? = withContext(Dispatchers.IO) {
        try {
            val url = "https://api.github.com/repos/$owner/$repo/contents/$path"
            client.newCall(authRequest(url, token)).execute().use { response ->
                if (response.isSuccessful) {
                    return@withContext JSONObject(response.body?.string() ?: "{}").optString("sha")
                }
            }
        } catch (e: Exception) {}
        return@withContext null
    }

    suspend fun updateFile(token: String, owner: String, repo: String, path: String, newContent: String, commitMessage: String): Boolean = withContext(Dispatchers.IO) {
        try {
            val sha = getFileSha(token, owner, repo, path)
            val payload = JSONObject().apply {
                put("message", commitMessage)
                put("content", Base64.getEncoder().encodeToString(newContent.toByteArray()))
                if (sha != null) put("sha", sha)
            }
            
            val request = Request.Builder()
                .url("https://api.github.com/repos/$owner/$repo/contents/$path")
                .header("Authorization", "Bearer $token")
                .header("Accept", "application/vnd.github.v3+json")
                .put(payload.toString().toRequestBody(jsonMediaType))
                .build()

            client.newCall(request).execute().use { response -> return@withContext response.isSuccessful }
        } catch (e: Exception) { return@withContext false }
    }

    // دالة حذف الملفات
    suspend fun deleteFile(token: String, owner: String, repo: String, path: String, commitMessage: String): Boolean = withContext(Dispatchers.IO) {
        try {
            val sha = getFileSha(token, owner, repo, path) ?: return@withContext false
            val payload = JSONObject().apply {
                put("message", commitMessage)
                put("sha", sha)
            }
            
            val request = Request.Builder()
                .url("https://api.github.com/repos/$owner/$repo/contents/$path")
                .header("Authorization", "Bearer $token")
                .header("Accept", "application/vnd.github.v3+json")
                .delete(payload.toString().toRequestBody(jsonMediaType))
                .build()

            client.newCall(request).execute().use { response -> return@withContext response.isSuccessful }
        } catch (e: Exception) { return@withContext false }
    }
}

package com.hiyako.app.util

import android.content.Context
import android.net.Uri
import android.provider.OpenableColumns

fun queryDisplayName(context: Context, uri: Uri): String {
    var name: String? = null
    try {
        context.contentResolver.query(uri, arrayOf(OpenableColumns.DISPLAY_NAME), null, null, null)?.use { cursor ->
            if (cursor.moveToFirst()) {
                val index = cursor.getColumnIndex(OpenableColumns.DISPLAY_NAME)
                if (index >= 0) name = cursor.getString(index)
            }
        }
    } catch (e: Exception) {
        // ignore and fall back below
    }
    return name ?: uri.lastPathSegment ?: "file"
}

fun isRtlText(text: String): Boolean {
    for (ch in text) {
        if (Character.isLetter(ch)) {
            val direction = Character.getDirectionality(ch)
            return direction == Character.DIRECTIONALITY_RIGHT_TO_LEFT ||
                direction == Character.DIRECTIONALITY_RIGHT_TO_LEFT_ARABIC
        }
    }
    return false
}
