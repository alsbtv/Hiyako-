package com.hiyako.app

import android.app.NotificationChannel
import android.app.NotificationManager
import android.app.Service
import android.content.Intent
import android.os.Build
import android.os.IBinder
import android.os.PowerManager
import androidx.core.app.NotificationCompat

class NexusService : Service() {
    private var wakeLock: PowerManager.WakeLock? = null

    override fun onBind(intent: Intent?): IBinder? = null

    override fun onCreate() {
        super.onCreate()
        val powerManager = getSystemService(POWER_SERVICE) as PowerManager
        wakeLock = powerManager.newWakeLock(PowerManager.PARTIAL_WAKE_LOCK, "NexusAI::ProcessingWakeLock")
        wakeLock?.acquire(10 * 60 * 1000L) // 10 minutes max
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        val channelId = "nexus_fg_channel"
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val channel = NotificationChannel(channelId, "Nexus AI Processing", NotificationManager.IMPORTANCE_LOW)
            getSystemService(NotificationManager::class.java).createNotificationChannel(channel)
        }

        val notification = NotificationCompat.Builder(this, channelId)
            .setContentTitle("Nexus AI is working...")
            .setContentText("Generating response in the background")
            .setSmallIcon(android.R.drawable.ic_dialog_info)
            .setOngoing(true)
            .build()

        startForeground(1, notification)
        return START_NOT_STICKY
    }

    override fun onDestroy() {
        wakeLock?.let { if (it.isHeld) it.release() }
        super.onDestroy()
    }
}

package com.hiyako.app.engine

import android.annotation.SuppressLint
import android.content.Context
import android.os.Message
import android.webkit.JavascriptInterface
import android.webkit.WebChromeClient
import android.webkit.WebView
import android.webkit.WebViewClient
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import org.json.JSONObject

interface PuterListener {
    fun onEngineReady()
    fun onModelsReceived(json: String)
    fun onChunk(reqId: String, text: String)
    fun onDone(reqId: String)
    fun onError(reqId: String, message: String)
    fun onTitle(convoId: Int, title: String)
    fun onAuth(signedIn: Boolean, username: String)
    fun provideImage(reqId: String): String
}

/**
 * Owns the hidden WebView that runs Puter.js. It lives as long as the ViewModel,
 * so switching screens never destroys the page or an in-flight request.
 */
class PuterEngine(context: Context, private val listener: PuterListener) {

    private val appContext: Context = context.applicationContext

    private val _ready = MutableStateFlow(false)
    val ready: StateFlow<Boolean> = _ready.asStateFlow()

    private val _popup = MutableStateFlow<WebView?>(null)
    val popup: StateFlow<WebView?> = _popup.asStateFlow()

    val webView: WebView

    init {
        webView = createWebView()
    }

    @SuppressLint("SetJavaScriptEnabled")
    private fun createWebView(): WebView {
        val wv = WebView(appContext)
        wv.setBackgroundColor(android.graphics.Color.TRANSPARENT)
        wv.settings.javaScriptEnabled = true
        wv.settings.domStorageEnabled = true
        wv.settings.setSupportMultipleWindows(true)
        wv.settings.javaScriptCanOpenWindowsAutomatically = true
        wv.webViewClient = WebViewClient()
        wv.webChromeClient = object : WebChromeClient() {
            override fun onCreateWindow(
                view: WebView?,
                isDialog: Boolean,
                isUserGesture: Boolean,
                resultMsg: Message?
            ): Boolean {
                if (resultMsg == null) return false
                val popupView = WebView(appContext)
                popupView.settings.javaScriptEnabled = true
                popupView.settings.domStorageEnabled = true
                popupView.settings.javaScriptCanOpenWindowsAutomatically = true
                popupView.webViewClient = WebViewClient()
                popupView.webChromeClient = object : WebChromeClient() {
                    override fun onCloseWindow(window: WebView?) {
                        closePopup()
                    }
                }
                val transport = resultMsg.obj as WebView.WebViewTransport
                transport.webView = popupView
                resultMsg.sendToTarget()
                _popup.value = popupView
                return true
            }
        }
        wv.addJavascriptInterface(Bridge(), "AndroidBridge")
        wv.loadDataWithBaseURL(BASE_URL, PAGE_HTML, "text/html", "UTF-8", null)
        return wv
    }

    fun closePopup() {
        val p = _popup.value
        _popup.value = null
        if (p != null) {
            p.post {
                try {
                    p.stopLoading()
                    p.loadUrl("about:blank")
                } catch (e: Exception) {
                    // ignore
                }
            }
        }
    }

    fun ask(reqId: String, messagesJson: String, model: String, hasImage: Boolean) {
        val js = "window.ask(" + JSONObject.quote(reqId) + "," + JSONObject.quote(messagesJson) + "," +
            JSONObject.quote(model) + "," + hasImage + ");"
        run(js)
    }

    fun makeTitle(convoId: Int, text: String) {
        run("window.makeTitle(" + convoId + "," + JSONObject.quote(text) + ");")
    }

    fun refreshModels() {
        run("window.fetchModels();")
    }

    fun checkAuth() {
        run("window.checkAuth();")
    }

    fun signIn() {
        run("window.signIn();")
    }

    fun reload() {
        _ready.value = false
        webView.post { webView.loadDataWithBaseURL(BASE_URL, PAGE_HTML, "text/html", "UTF-8", null) }
    }

    fun destroy() {
        try {
            webView.destroy()
        } catch (e: Exception) {
            // ignore
        }
    }

    private fun run(js: String) {
        webView.post { webView.evaluateJavascript(js, null) }
    }

    inner class Bridge {
        @JavascriptInterface
        fun onReady() {
            _ready.value = true
            listener.onEngineReady()
        }

        @JavascriptInterface
        fun onModels(json: String) {
            listener.onModelsReceived(json)
        }

        @JavascriptInterface
        fun onChunk(reqId: String, text: String) {
            listener.onChunk(reqId, text)
        }

        @JavascriptInterface
        fun onDone(reqId: String) {
            listener.onDone(reqId)
        }

        @JavascriptInterface
        fun onError(reqId: String, message: String) {
            listener.onError(reqId, message)
        }

        @JavascriptInterface
        fun onTitle(convoId: String, title: String) {
            val id = convoId.toIntOrNull() ?: return
            listener.onTitle(id, title)
        }

        @JavascriptInterface
        fun onAuth(signedIn: String, username: String) {
            listener.onAuth(signedIn == "true", username)
        }

        @JavascriptInterface
        fun getImage(reqId: String): String {
            return listener.provideImage(reqId)
        }
    }

    companion object {
        // Keep this base URL identical to the old version so an existing Puter session in localStorage survives.
        private const val BASE_URL = "https://appassets.androidplatform.net"

        private val PAGE_HTML = """
            <html><head><meta charset='utf-8'><script src='https://js.puter.com/v2/'></script></head><body>
            <script>
            function errText(e) {
                try {
                    if (!e) return 'Unknown error';
                    if (typeof e === 'string') return e;
                    if (e.message) return e.message;
                    if (e.error && e.error.message) return e.error.message;
                    if (e.error && typeof e.error === 'string') return e.error;
                    return JSON.stringify(e);
                } catch (x) { return 'Unknown error'; }
            }
            function contentText(c) {
                if (c === null || c === undefined) return '';
                if (typeof c === 'string') return c;
                if (Array.isArray(c)) {
                    return c.map(function (p) { return (p && p.text) ? p.text : (typeof p === 'string' ? p : ''); }).join('');
                }
                if (c.text) return c.text;
                return JSON.stringify(c);
            }
            function respText(r) {
                if (r === null || r === undefined) return '';
                if (typeof r === 'string') return r;
                if (r.message) return contentText(r.message.content);
                if (r.text) return r.text;
                return JSON.stringify(r);
            }
            window.fetchModels = function () {
                try {
                    puter.ai.listModels().then(function (models) {
                        var result = models.map(function (m) {
                            if (typeof m === 'string') return { id: m, name: m };
                            return { id: m.id, name: m.name || m.id };
                        });
                        AndroidBridge.onModels(JSON.stringify(result));
                    }).catch(function (e) {});
                } catch (e) {}
            };
            window.ask = async function (reqId, msgsJson, model, hasImage) {
                try {
                    var msgs = JSON.parse(msgsJson);
                    var opts = {};
                    if (model) { opts.model = model; }
                    var resp;
                    if (hasImage) {
                        var img = AndroidBridge.getImage(reqId);
                        var prompt = contentText(msgs[msgs.length - 1].content);
                        resp = await puter.ai.chat(prompt, img, false, opts);
                    } else {
                        opts.stream = true;
                        resp = await puter.ai.chat(msgs, false, opts);
                    }
                    if (resp && resp.success === false) { throw resp; }
                    if (resp && typeof resp[Symbol.asyncIterator] === 'function') {
                        for await (var part of resp) {
                            if (part && part.text) { AndroidBridge.onChunk(reqId, part.text); }
                        }
                    } else {
                        var t = respText(resp);
                        if (t) { AndroidBridge.onChunk(reqId, t); }
                    }
                    AndroidBridge.onDone(reqId);
                } catch (e) {
                    AndroidBridge.onError(reqId, errText(e));
                }
            };
            window.makeTitle = function (convoId, text) {
                try {
                    var p = 'Write a short title (2 to 5 words) for a conversation that starts with the message below. Use the same language as the message. Output ONLY the title, with no quotes and no trailing punctuation.\n\nMessage: ' + String(text).substring(0, 300);
                    puter.ai.chat(p).then(function (r) {
                        var t = respText(r);
                        if (t) { AndroidBridge.onTitle(String(convoId), t.trim()); }
                    }).catch(function (e) {});
                } catch (e) {}
            };
            window.checkAuth = function () {
                try {
                    if (puter.auth.isSignedIn()) {
                        puter.auth.getUser().then(function (u) {
                            AndroidBridge.onAuth('true', (u && u.username) ? u.username : '');
                        }).catch(function (e) { AndroidBridge.onAuth('true', ''); });
                    } else {
                        AndroidBridge.onAuth('false', '');
                    }
                } catch (e) { AndroidBridge.onAuth('false', ''); }
            };
            window.signIn = function () {
                try {
                    puter.auth.signIn().then(function () {
                        window.checkAuth();
                        window.fetchModels();
                    }).catch(function (e) {});
                } catch (e) {}
            };
            (function boot() {
                if (typeof puter === 'undefined') { setTimeout(boot, 300); return; }
                AndroidBridge.onReady();
                window.fetchModels();
                window.checkAuth();
            })();
            </script></body></html>
        """.trimIndent()
    }
}

package com.hiyako.app.ui.i18n

import android.content.Context
import androidx.compose.runtime.getValue
import androidx.compose.runtime.mutableStateOf
import androidx.compose.runtime.setValue

/** The app language. Arabic is the default; English is available in Settings. */
object Lang {
    private const val PREFS = "nexus_settings"
    private const val KEY = "app_lang"

    var arabic: Boolean by mutableStateOf(true)

    fun load(context: Context) {
        arabic = try {
            context.getSharedPreferences(PREFS, Context.MODE_PRIVATE).getString(KEY, "ar") != "en"
        } catch (e: Exception) {
            true
        }
    }

    fun set(context: Context, useArabic: Boolean) {
        arabic = useArabic
        try {
            context.getSharedPreferences(PREFS, Context.MODE_PRIVATE)
                .edit()
                .putString(KEY, if (useArabic) "ar" else "en")
                .apply()
        } catch (e: Exception) {
            // Preferences unavailable: the language still changes for this session.
        }
    }
}

/** Returns the English or Arabic text for the current language. Reading it inside composables keeps them in sync. */
fun tr(en: String, ar: String): String = if (Lang.arabic) ar else en

private sealed class MdBlock {
    class Para(val text: String) : MdBlock()
    class Head(val level: Int, val text: String) : MdBlock()
    class Bullets(val items: List<String>) : MdBlock()
    class Numbers(val items: List<String>) : MdBlock()
    class Code(val lang: String, val code: String) : MdBlock()
    class Quote(val text: String) : MdBlock()
}

private val numberedRegex = Regex("^\\d+[.)]\\s+")
private val headingRegex = Regex("^(#{1,6})\\s+(.*)$")

private fun parseMarkdownBlocks(src: String): List<MdBlock> {
    val out = ArrayList<MdBlock>()
    val para = ArrayList<String>()
    val bullets = ArrayList<String>()
    val numbers = ArrayList<String>()
    val quote = ArrayList<String>()

    fun flush() {
        if (para.isNotEmpty()) { out.add(MdBlock.Para(para.joinToString("\n"))); para.clear() }
        if (bullets.isNotEmpty()) { out.add(MdBlock.Bullets(bullets.toList())); bullets.clear() }
        if (numbers.isNotEmpty()) { out.add(MdBlock.Numbers(numbers.toList())); numbers.clear() }
        if (quote.isNotEmpty()) { out.add(MdBlock.Quote(quote.joinToString("\n"))); quote.clear() }
    }

    val lines = src.lines()
    var i = 0
    while (i < lines.size) {
        val trimmed = lines[i].trim()
        val heading = headingRegex.find(trimmed)
        when {
            trimmed.startsWith("```") -> {
                flush()
                val lang = trimmed.removePrefix("```").trim()
                val code = ArrayList<String>()
                i++
                while (i < lines.size && !lines[i].trim().startsWith("```")) { code.add(lines[i]); i++ }
                out.add(MdBlock.Code(lang, code.joinToString("\n")))
            }
            trimmed.isEmpty() -> flush()
            heading != null -> {
                flush()
                out.add(MdBlock.Head(heading.groupValues[1].length, heading.groupValues[2].trim()))
            }
            trimmed.startsWith("- ") || trimmed.startsWith("* ") || trimmed.startsWith("• ") -> {
                if (bullets.isEmpty()) flush()
                bullets.add(trimmed.substring(2).trim())
            }
            numberedRegex.containsMatchIn(trimmed) -> {
                if (numbers.isEmpty()) flush()
                numbers.add(trimmed.replace(numberedRegex, "").trim())
            }
            trimmed.startsWith("> ") -> {
                if (quote.isEmpty()) flush()
                quote.add(trimmed.substring(2).trim())
            }
            else -> {
                if (para.isEmpty()) flush()
                para.add(lines[i])
            }
        }
        i++
    }
    flush()
    return out
}

private fun inlineStyled(text: String): AnnotatedString = buildAnnotatedString {
    var i = 0
    while (i < text.length) {
        if (text.startsWith("**", i)) {
            val end = text.indexOf("**", i + 2)
            if (end > i + 2) {
                withStyle(SpanStyle(fontWeight = FontWeight.SemiBold)) { append(text.substring(i + 2, end)) }
                i = end + 2
            } else { append("**"); i += 2 }
        } else if (text[i] == '`') {
            val end = text.indexOf('`', i + 1)
            if (end > i + 1) {
                withStyle(SpanStyle(fontFamily = FontFamily.Monospace, color = AccentBlue)) { append(text.substring(i + 1, end)) }
                i = end + 1
            } else { append('`'); i++ }
        } else {
            append(text[i]); i++
        }
    }
}

@Composable
fun MarkdownText(text: String) {
    val blocks = remember(text) { parseMarkdownBlocks(text) }
    Column(verticalArrangement = Arrangement.spacedBy(8.dp)) {
        blocks.forEach { block ->
            when (block) {
                is MdBlock.Para -> Text(inlineStyled(block.text), color = TextPrimary, fontSize = 16.sp, lineHeight = 26.sp, style = LocalTextStyle.current.copy(textDirection = TextDirection.Content))
                is MdBlock.Head -> Text(inlineStyled(block.text), color = TextPrimary, fontSize = (28 - block.level * 2).sp, fontWeight = FontWeight.Bold, lineHeight = 30.sp)
                is MdBlock.Bullets -> Column(verticalArrangement = Arrangement.spacedBy(4.dp)) {
                    block.items.forEach { item ->
                        Row {
                            Text("•  ", color = AccentBlue, fontSize = 16.sp, lineHeight = 26.sp)
                            Text(inlineStyled(item), color = TextPrimary, fontSize = 16.sp, lineHeight = 26.sp, modifier = Modifier.weight(1f), style = LocalTextStyle.current.copy(textDirection = TextDirection.Content))
                        }
                    }
                }
                is MdBlock.Numbers -> Column(verticalArrangement = Arrangement.spacedBy(4.dp)) {
                    block.items.forEachIndexed { index, item ->
                        Row {
                            Text("${index + 1}.  ", color = AccentBlue, fontSize = 16.sp, lineHeight = 26.sp)
                            Text(inlineStyled(item), color = TextPrimary, fontSize = 16.sp, lineHeight = 26.sp, modifier = Modifier.weight(1f), style = LocalTextStyle.current.copy(textDirection = TextDirection.Content))
                        }
                    }
                }
                is MdBlock.Quote -> Row(modifier = Modifier.height(IntrinsicSize.Min)) {
                    Box(Modifier.width(3.dp).fillMaxHeight().background(AccentBlue, RoundedCornerShape(2.dp)))
                    Spacer(Modifier.width(10.dp))
                    Text(inlineStyled(block.text), color = TextSecondary, fontSize = 16.sp, lineHeight = 26.sp, modifier = Modifier.weight(1f))
                }
                is MdBlock.Code -> CollapsibleCodeBlock(block.lang, block.code)
            }
        }
    }
}

    val progress = remember { Animatable(0f) }

import kotlinx.coroutines.launch

class WorkspaceViewModel : ViewModel() {
    private val _repos = MutableStateFlow<List<Repo>>(emptyList())
    val repos = _repos.asStateFlow()

    private val _isLoading = MutableStateFlow(false)
    val isLoading = _isLoading.asStateFlow()
    
    private val _progress = MutableStateFlow(0f)
    val progress = _progress.asStateFlow()

    private val _loadingMessage = MutableStateFlow("")
    val loadingMessage = _loadingMessage.asStateFlow()

    private val _token = MutableStateFlow("")
    val token = _token.asStateFlow()

    private val _error = MutableStateFlow<String?>(null)
    val error = _error.asStateFlow()

    fun loadToken(context: Context) {
        val vault = KeyVault.prefs(context)
        val legacy = context.getSharedPreferences("nexus_prefs", Context.MODE_PRIVATE)
        val oldToken = legacy.getString("github_token", "").orEmpty()
        if (oldToken.isNotBlank() && vault.getString("github_token_enc", null) == null) {
            vault.edit().putString("github_token_enc", KeyVault.encrypt(oldToken)).apply()
        }
        if (legacy.contains("github_token") && vault.getString("github_token_enc", null) != null) {
            legacy.edit().remove("github_token").apply()
        }
        val enc = KeyVault.prefs(context).getString("github_token_enc", null)
        _token.value = if (enc != null) KeyVault.decrypt(enc).orEmpty() else ""
        if (_token.value.isNotBlank()) fetchRepos()
    }

    fun saveToken(context: Context, token: String) {
        val clean = token.trim()
        if (clean.isNotEmpty()) {
            val enc = KeyVault.encrypt(clean)
            KeyVault.prefs(context).edit().putString("github_token_enc", enc).apply()
        } else {
            KeyVault.prefs(context).edit().remove("github_token_enc").apply()
        }
        _token.value = clean
        fetchRepos()
    }

    fun disconnect(context: Context) {
        KeyVault.prefs(context).edit().remove("github_token_enc").apply()
        context.getSharedPreferences("nexus_prefs", Context.MODE_PRIVATE).edit().remove("github_token").apply()
        WorkspaceManager.activeRepo = ""
        WorkspaceManager.projectFiles = emptyMap()
        WorkspaceManager.isLoaded = false
        _token.value = ""
        _repos.value = emptyList()
        _error.value = null
    }

    fun fetchRepos() {
        if (_token.value.isBlank()) return
        _isLoading.value = true
        _error.value = null
        viewModelScope.launch {
            val res = GitHubApi.getUserRepos(_token.value)
            _repos.value = res
            if (res.isEmpty()) _error.value = "Could not load repositories. Check your token."
            _isLoading.value = false
        }
    }

    fun selectRepoAndLoadFiles(repo: Repo, onComplete: () -> Unit) {
        _isLoading.value = true
        _error.value = null
        _progress.value = 0f
        _loadingMessage.value = "Fetching file tree..."
        viewModelScope.launch {
            val owner = repo.fullName.split("/")[0]
            val filesList = GitHubApi.getRepoFiles(_token.value, owner, repo.name, repo.defaultBranch)
            if (filesList.isEmpty()) {
                _isLoading.value = false
                _error.value = "No supported files found in repo."
                return@launch
            }
            
            val loadedFiles = java.util.concurrent.ConcurrentHashMap<String, String>()
            val finished = java.util.concurrent.atomic.AtomicInteger(0)
            val gate = Semaphore(8)
            val tokenValue = _token.value
            coroutineScope {
                filesList.map { path ->
                    async(Dispatchers.IO) {
                        gate.withPermit {
                            val content = GitHubApi.getFileContent(tokenValue, owner, repo.name, path)
                            if (content != null) loadedFiles[path] = content
                            val count = finished.incrementAndGet()
                            _progress.value = count.toFloat() / filesList.size
                            _loadingMessage.value = "$count / ${filesList.size}"
                        }
                    }
                }.awaitAll()
            }
            
            WorkspaceManager.activeRepo = repo.fullName
            WorkspaceManager.projectFiles = loadedFiles.toMap()
            WorkspaceManager.isLoaded = loadedFiles.isNotEmpty()
            
            _isLoading.value = false
            _loadingMessage.value = ""
            if (loadedFiles.isNotEmpty()) onComplete()
            else _error.value = "Failed to download file contents."
        }
    }
}

    var statusText by remember { mutableStateOf(tr("Click below to fetch the latest APK from GitHub.", "اضغط بالأسفل لجلب أحدث نسخة من GitHub.")) }

                text = tr("Download & Install", "تنزيل وتثبيت"),
                onClick = {
                    if (!context.packageManager.canRequestPackageInstalls()) {
                        context.startActivity(Intent(Settings.ACTION_MANAGE_UNKNOWN_APP_SOURCES, Uri.parse("package:${context.packageName}")))
                        return@HiyakoButton
                    }
                    isDownloading = true
                    statusText = tr("Downloading update...", "جارٍ التنزيل...")
                    scope.launch(Dispatchers.IO) {
                        try {

                            val response = client.newCall(Request.Builder().url(apkUrl).build()).execute()
                            response.use { resp ->
                                if (!resp.isSuccessful) error("HTTP ${resp.code}")
                                val body = resp.body ?: error("Empty body")
                                if (body.contentLength() <= 0) error("Invalid APK size")
                                val file = File(context.cacheDir, "update.apk")
                                body.byteStream().use { input -> file.outputStream().use { output -> input.copyTo(output) } }
                                withContext(Dispatchers.Main) {
                                    statusText = tr("Download complete. Installing...", "اكتمل التنزيل. جارٍ التثبيت...")
                                    val intent = Intent(Intent.ACTION_VIEW).apply {
                                        setDataAndType(androidx.core.content.FileProvider.getUriForFile(context, "${context.packageName}.provider", file), "application/vnd.android.package-archive")
                                        addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION or Intent.FLAG_ACTIVITY_NEW_TASK)
                                    }
                                    try { context.startActivity(intent) } catch (e: Exception) { statusText = "No installer found." }
                                    isDownloading = false
                                }
                            }
                        } catch (e: Exception) {
                            withContext(Dispatchers.Main) { statusText = "Error: ${e.message}"; isDownloading = false }
                        }
                    }
                },

                modifier = Modifier.fillMaxWidth()
            )

    }

private fun dateBucket(ts: Long, now: Long = System.currentTimeMillis()): Int {
    val c1 = Calendar.getInstance().apply { timeInMillis = ts; set(Calendar.HOUR_OF_DAY, 0); set(Calendar.MINUTE, 0); set(Calendar.SECOND, 0); set(Calendar.MILLISECOND, 0) }
    val c2 = Calendar.getInstance().apply { timeInMillis = now; set(Calendar.HOUR_OF_DAY, 0); set(Calendar.MINUTE, 0); set(Calendar.SECOND, 0); set(Calendar.MILLISECOND, 0) }
    val days = (c2.timeInMillis - c1.timeInMillis) / 86_400_000L
    return when { days <= 0 -> 0; days == 1L -> 1; days <= 7 -> 2; else -> 3 }
}

    private fun cancelRequestsFor(convoId: Int) {

    fun sendMessage(text: String, attachmentUri: Uri?) {
        val convoId = _activeConversationId.value ?: return
        val trimmed = text.trim()
        if (trimmed.isEmpty() && attachmentUri == null) return
        if (!markGenerating(convoId)) return

        val currentTitle = _conversations.value.firstOrNull { it.id == convoId }?.title
        val needsTitle = currentTitle == null || currentTitle == DEFAULT_TITLE || currentTitle == "New Group"

        viewModelScope.launch {
            try {
                var type = "TEXT"; var storedUri: String? = null; var displayName: String? = null
                if (attachmentUri != null) {
                    val saved = withContext(Dispatchers.IO) { copyAttachment(attachmentUri) }
                    type = if (saved.mime.startsWith("image/")) "IMAGE" else "FILE"
                    storedUri = Uri.fromFile(saved.file).toString(); displayName = saved.name
                }
                chatDao.insertMessage(ChatEntity(conversationId = convoId, role = "user", content = trimmed, messageType = type, attachmentUri = storedUri, senderName = displayName))
                conversationDao.touch(convoId, System.currentTimeMillis())
                if (needsTitle) {
                    val seed = if (trimmed.isNotEmpty()) trimmed else (displayName ?: "Attachment")
                    val fallback = seed.lines().first().trim().take(32).ifBlank { DEFAULT_TITLE }
                    titleFallbacks[convoId] = fallback
                    conversationDao.updateTitle(convoId, fallback)
                    engine.makeTitle(convoId, seed)
                }
            } catch (e: Exception) {
                _generating.update { it - convoId }; _errors.update { it + (convoId to ("Could not attach the file: " + (e.message ?: "unknown error"))) }; return@launch
            }
            runRequest(convoId)
        }
    }

    fun regenerate() {
        val convoId = _activeConversationId.value ?: return
        if (!markGenerating(convoId)) return
        viewModelScope.launch {
            try { val last = chatDao.getMessagesOnce(convoId).lastOrNull(); if (last != null && last.role != "user") chatDao.deleteMessage(last.id) }
            catch (e: Exception) { _generating.update { it - convoId }; return@launch }
            runRequest(convoId)
        }
    }

    fun stopGeneration() {
        val convoId = _activeConversationId.value ?: return
        val partials = mutableListOf<String>()
        pending.entries.toList().forEach { entry -> if (entry.value.convoId == convoId) { pending.remove(entry.key); pendingImages.remove(entry.key); partials.add(entry.value.text()) } }
        val partial = partials.joinToString("").trim()
        viewModelScope.launch {
            if (partial.isNotEmpty()) chatDao.insertMessage(ChatEntity(conversationId = convoId, role = "ai", content = partial))
            _generating.update { it - convoId }; stopServiceIfIdle()
        }
    }

    private suspend fun runRequest(convoId: Int) {
        requestCounter += 1
        val reqId = "r$requestCounter"
        pending[reqId] = Pending(convoId)
        startServiceSafely()
        try {
            val entities = chatDao.getMessagesOnce(convoId)
            val built = withContext(Dispatchers.IO) { buildRequest(entities) }
            if (!pending.containsKey(reqId)) return
            val image = built.imageDataUrl
            if (image != null) pendingImages[reqId] = image
            val selected = _selectedModel.value
            if (selected.accountId.isNotEmpty()) streamViaAccount(reqId, selected.accountId, selected.id, built.messagesJson, image)
            else { pendingPayloads[reqId] = built.messagesJson to image; engine.ask(reqId, built.messagesJson, selected.id, image != null) }
        } catch (e: Exception) { failRequest(reqId, e.message ?: "Unexpected error") }
    }

    fun reloadAccountModels() { _availableModels.update { list -> list.filter { it.accountId.isEmpty() } + accountModelsOf(appCtx) } }

    private fun streamViaAccount(reqId: String, accountId: String, modelId: String, messagesJson: String, imageDataUrl: String?) {
        viewModelScope.launch(Dispatchers.IO) {
            val accounts = AccountStore.load(appCtx)
            val start = accounts.indexOfFirst { it.id == accountId }
            if (start < 0) { failRequest(reqId, "This account was removed. Pick another model."); return@launch }
            val messages = try { buildAccountMessages(messagesJson, imageDataUrl) } catch (e: Exception) { failRequest(reqId, "Could not prepare the message."); return@launch }
            val order = accounts.subList(start, accounts.size) + accounts.subList(0, start)
            var lastError: Throwable? = null
            for (account in order) {
                if (!pending.containsKey(reqId)) return@launch
                val spec = ProviderCatalog.find(account.providerId) ?: continue
                if (spec.id == ProviderCatalog.PUTER) continue
                val result = OpenAiCompatClient.stream(
                    baseUrl = account.baseUrl, apiKey = AccountStore.apiKey(account),
                    model = if (account.id == accountId) modelId else defaultModelOf(appCtx, account),
                    messages = messages
                ) { delta -> if (!pending.containsKey(reqId)) throw CancellationException("Stopped") else onChunk(reqId, delta) }
                if (result.isSuccess) { onDone(reqId); return@launch }
                lastError = result.exceptionOrNull()
                if (lastError == null || !OpenAiCompatClient.isQuotaError(lastError)) break
            }
            failRequest(reqId, lastError?.message ?: "No saved account could answer this request.")
        }
    }

    private fun buildAccountMessages(messagesJson: String, imageDataUrl: String?): JSONArray {
        val array = JSONArray(messagesJson)
        if (imageDataUrl != null) {
            for (i in array.length() - 1 downTo 0) {
                val message = array.getJSONObject(i)
                if (message.optString("role") == "user") {
                    val parts = JSONArray().put(JSONObject().put("type", "text").put("text", message.optString("content")))
                        .put(JSONObject().put("type", "image_url").put("image_url", JSONObject().put("url", imageDataUrl)))
                    message.put("content", parts); break
                }
            }
        }
        return array
    }

    private fun buildRequest(entities: List<ChatEntity>): BuiltRequest {
        val system = StringBuilder()
        system.append("You are Hiyako, a helpful assistant. Reply in the same language as the user's last message. Use Markdown for formatting and fenced code blocks for code.")
        val mem = _memories.value.take(20)
        if (mem.isNotEmpty()) { system.append("\n\nKnown facts about the user:\n"); mem.forEach { system.append("- ").append(it.content.take(300)).append('\n') } }
        
        if (WorkspaceManager.isLoaded && prefs.getBoolean("use_workspace_context", false)) {
            system.append("\n\nActive repository: ").append(WorkspaceManager.activeRepo).append('\n')
            var budget = 60000
            for ((path, content) in WorkspaceManager.projectFiles.toSortedMap()) {
                if (budget <= 0) break
                val chunk = content.take(minOf(budget, 6000))
                system.append("\n=== ").append(path).append(" ===\n").append(chunk).append('\n')
                budget -= chunk.length
            }
        }

        if (isThinkingMode) system.append("\n\nThink through the problem carefully and step by step before giving the final answer.")
        if (isSearchMode) system.append("\n\nThe user turned on Search mode, but you cannot browse the web in this app. If the answer depends on recent information, say so clearly instead of guessing.")

        val arr = JSONArray()
        arr.put(JSONObject().put("role", "system").put("content", system.toString()))
        val recent = entities.filterNot { it.role == "ai" && it.content == "Thinking..." }.takeLast(30)

        for (e in recent) {
            val role = if (e.role == "user") "user" else "assistant"
            var content = e.content.take(12000)
            if (role == "user" && e.messageType == "FILE") {
                val name = e.senderName ?: "file"
                val fileText = readTextAttachment(e.attachmentUri)
                content += if (fileText != null) "\n\n[Attached file: $name]\n```\n$fileText\n```" else "\n\n[Attached file: $name - this file type cannot be read in this version of the app]"
            }
            arr.put(JSONObject().put("role", role).put("content", content))
        }

        var imageDataUrl: String? = null
        val last = recent.lastOrNull()
        if (last != null && last.role == "user" && last.messageType == "IMAGE") {
            val path = last.attachmentUri?.let { Uri.parse(it).path }
            if (path != null) imageDataUrl = imageToDataUrl(File(path))
            if (imageDataUrl != null && last.content.isBlank()) {
                arr.remove(arr.length() - 1); arr.put(JSONObject().put("role", "user").put("content", "Describe this image in detail."))
            }
        }
        return BuiltRequest(arr.toString(), imageDataUrl)
    }

    private fun copyAttachment(uri: Uri): SavedAttachment {
        val name = queryDisplayName(appCtx, uri)
        val mime = appCtx.contentResolver.getType(uri) ?: "application/octet-stream"
        val dir = File(appCtx.filesDir, "attachments").apply { mkdirs() }
        val safe = name.replace(Regex("[^A-Za-z0-9._-]"), "_")
        val out = File(dir, System.currentTimeMillis().toString() + "_" + safe)
        appCtx.contentResolver.openInputStream(uri)?.use { src -> out.outputStream().use { dst -> src.copyTo(dst) } } ?: throw IllegalStateException("Cannot open file")
        return SavedAttachment(out, mime, name)
    }

    private fun readTextAttachment(uriString: String?): String? {
        if (uriString == null) return null
        val path = Uri.parse(uriString).path ?: return null
        val file = File(path)
        if (!file.exists() || file.length() > 5_000_000L) return null
        val bytes = file.readBytes()
        for (b in bytes) { if (b.toInt() == 0) return null }
        return String(bytes, Charsets.UTF_8).take(40000)
    }

    private fun imageToDataUrl(file: File): String? {
        if (!file.exists()) return null
        val bounds = BitmapFactory.Options().apply { inJustDecodeBounds = true }; BitmapFactory.decodeFile(file.absolutePath, bounds)
        var sample = 1; while (bounds.outWidth / sample > 1600 || bounds.outHeight / sample > 1600) sample *= 2
        val opts = BitmapFactory.Options().apply { inSampleSize = sample }
        val bitmap = BitmapFactory.decodeFile(file.absolutePath, opts) ?: return null
        val out = ByteArrayOutputStream()
        val flat = if (bitmap.hasAlpha()) {
            val background = Bitmap.createBitmap(bitmap.width, bitmap.height, Bitmap.Config.ARGB_8888)
            val canvas = android.graphics.Canvas(background)
            canvas.drawColor(android.graphics.Color.WHITE)
            canvas.drawBitmap(bitmap, 0f, 0f, null)
            bitmap.recycle()
            background
        } else bitmap
        flat.compress(Bitmap.CompressFormat.JPEG, 85, out); flat.recycle()
        return "data:image/jpeg;base64," + Base64.encodeToString(out.toByteArray(), Base64.NO_WRAP)
    }

    private fun startServiceSafely() { try { ContextCompat.startForegroundService(appCtx, Intent(appCtx, NexusService::class.java)) } catch (e: Exception) {} }
    private fun stopServiceIfIdle() { viewModelScope.launch { delay(1500); if (pending.isEmpty()) { try { appCtx.stopService(Intent(appCtx, NexusService::class.java)) } catch (e: Exception) {} } } }
    private suspend fun conversationExists(id: Int): Boolean = _conversations.value.any { it.id == id }

    private fun failRequest(reqId: String, message: String) {
        val p = pending.remove(reqId) ?: return
        pendingImages.remove(reqId); val partial = p.text().trim()
        viewModelScope.launch {
            if (partial.isNotEmpty() && conversationExists(p.convoId)) chatDao.insertMessage(ChatEntity(conversationId = p.convoId, role = "ai", content = partial))
            _errors.update { it + (p.convoId to message) }; _generating.update { it - p.convoId }; stopServiceIfIdle()
        }
    }

    override fun onEngineReady() { _isEngineReady.value = true }
    
    override fun onModelsReceived(json: String) {
        try {
            val arr = JSONArray(json); val list = mutableListOf<AiModel>()
            for (i in 0 until arr.length()) { val o = arr.getJSONObject(i); val id = o.optString("id"); if (id.isNotBlank()) list.add(AiModel(o.optString("name", id).ifBlank { id }, id)) }
            val sorted = list.distinctBy { it.id }.sortedBy { it.name.lowercase() }
            _availableModels.value = listOf(AiModel("Default (Auto)", "")) + sorted + accountModelsOf(appCtx)
        } catch (e: Exception) {}
    }

    override fun onChunk(reqId: String, text: String) { val p = pending[reqId] ?: return; val snapshot = p.append(text); _generating.update { it + (p.convoId to snapshot) } }

    override fun onDone(reqId: String) {
        val p = pending.remove(reqId) ?: return
        pendingImages.remove(reqId); pendingPayloads.remove(reqId)
        val finalText = p.text().trim()
        viewModelScope.launch {
            if (conversationExists(p.convoId)) {
                if (finalText.isNotEmpty()) { chatDao.insertMessage(ChatEntity(conversationId = p.convoId, role = "ai", content = finalText)); conversationDao.touch(p.convoId, System.currentTimeMillis()) }
                else _errors.update { it + (p.convoId to "The model returned an empty reply.") }
            }
            _generating.update { it - p.convoId }; stopServiceIfIdle()
        }
    }

    override fun onError(reqId: String, message: String) {
        val payload = pendingPayloads.remove(reqId)
        val fallback = AccountStore.load(appCtx).firstOrNull { defaultModelOf(appCtx, it).isNotBlank() }
        val isQuota = OpenAiCompatClient.isQuotaError(Exception(message))
        if (payload != null && fallback != null && isQuota && pending.containsKey(reqId)) streamViaAccount(reqId, fallback.id, defaultModelOf(appCtx, fallback), payload.first, payload.second)
        else failRequest(reqId, message)
    }

    override fun onTitle(convoId: Int, title: String) {
        val clean = title.trim().trim('"', '\'', '«', '»').trim().take(60)
        if (clean.isEmpty()) return
        val fallback = titleFallbacks[convoId] ?: return
        val current = _conversations.value.firstOrNull { it.id == convoId }?.title
        if (current == fallback) { titleFallbacks.remove(convoId); viewModelScope.launch { conversationDao.updateTitle(convoId, clean) } }
    }

    override fun onAuth(signedIn: Boolean, username: String) { 
        _auth.value = AuthInfo(signedIn, username) 
        _isSigningIn.value = false // Stop loading when auth resolves
    }
    
    override fun provideImage(reqId: String): String = pendingImages[reqId] ?: ""

    override fun onCleared() { tts?.shutdown(); if (engineHolder.isInitialized()) engine.destroy(); super.onCleared() }

    companion object { private const val DEFAULT_TITLE = "New Chat" }
}

@Composable

    val mark = remember { Animatable(0f) }
    val name = remember { Animatable(0f) }
    val tagline = remember { Animatable(0f) }

        coroutineScope {
            launch { mark.animateTo(1f, tween(700, easing = FastOutSlowInEasing)) }
            launch { delay(350); name.animateTo(1f, tween(600, easing = FastOutSlowInEasing)) }

        delay(450)
        exit.animateTo(1f, tween(400))

        Modifier.fillMaxSize().graphicsLayer { alpha = 1f - exit.value },
        contentAlignment = Alignment.Center
    ) {
        Column(horizontalAlignment = Alignment.CenterHorizontally) {
            Box(
                Modifier
                    .size(96.dp)

                        alpha = mark.value

                        scaleY = 0.92f + 0.08f * mark.value
                    }

private fun wipeLocalData(context: Context) {
    Thread {
        try { AppDatabase.getDatabase(context).clearAllTables() } catch (e: Exception) {}
        KeyVault.prefs(context).edit().clear().apply()
        context.getSharedPreferences("nexus_settings", Context.MODE_PRIVATE).edit().clear().apply()

/** A saved account. The API key is stored only as an encrypted blob in keyEnc. */
data class Account(
    val id: String,
    val providerId: String,
    val label: String,
    val baseUrl: String,
    val model: String,
    val keyEnc: String
)

object AccountStore {
    private const val PREFS_KEY = "accounts_v1"

    fun load(context: Context): List<Account> {
        val raw = KeyVault.prefs(context).getString(PREFS_KEY, null) ?: return emptyList()
        return try {
            val array = JSONArray(raw)
            val result = ArrayList<Account>(array.length())
            for (i in 0 until array.length()) {
                val o = array.getJSONObject(i)
                result.add(
                    Account(
                        id = o.getString("id"),
                        providerId = o.getString("providerId"),
                        label = o.optString("label"),
                        baseUrl = o.optString("baseUrl"),
                        model = o.optString("model"),
                        keyEnc = o.optString("keyEnc")
                    )
                )
            }
            result
        } catch (e: Exception) {
            emptyList()
        }
    }

    fun save(context: Context, accounts: List<Account>) {
        val array = JSONArray()
        for (a in accounts) {
            array.put(
                JSONObject()
                    .put("id", a.id)
                    .put("providerId", a.providerId)
                    .put("label", a.label)
                    .put("baseUrl", a.baseUrl)
                    .put("model", a.model)
                    .put("keyEnc", a.keyEnc)
            )
        }
        KeyVault.prefs(context).edit().putString(PREFS_KEY, array.toString()).apply()
    }

    fun add(
        context: Context,
        providerId: String,
        label: String,
        baseUrl: String,
        model: String,
        apiKey: String
    ): Account {
        val account = Account(
            id = UUID.randomUUID().toString(),
            providerId = providerId,
            label = label,
            baseUrl = baseUrl.trim(),
            model = model.trim(),
            keyEnc = if (apiKey.isEmpty()) "" else KeyVault.encrypt(apiKey)
        )
        save(context, load(context) + account)
        return account
    }

    fun remove(context: Context, accountId: String) {
        save(context, load(context).filterNot { it.id == accountId })
        ModelCache.remove(context, accountId)
    }

    /** Returns the decrypted key, or an empty string when the account has none. Never log the result. */
    fun apiKey(account: Account): String {
        if (account.keyEnc.isEmpty()) return ""
        return KeyVault.decrypt(account.keyEnc).orEmpty()
    }
}

package com.hiyako.app.providers

import android.content.Context
import com.hiyako.app.security.KeyVault
import org.json.JSONArray

/** Model ids loaded from each account's /models endpoint, cached on the device. */
object ModelCache {
    private fun key(accountId: String) = "models_$accountId"

    fun models(context: Context, accountId: String): List<String> {
        val raw = KeyVault.prefs(context).getString(key(accountId), null) ?: return emptyList()
        return try {
            val array = JSONArray(raw)
            (0 until array.length()).map { array.getString(it) }
        } catch (e: Exception) {
            emptyList()
        }
    }

    fun save(context: Context, accountId: String, ids: List<String>) {
        KeyVault.prefs(context).edit().putString(key(accountId), JSONArray(ids).toString()).apply()
    }

    fun remove(context: Context, accountId: String) {
        KeyVault.prefs(context).edit().remove(key(accountId)).apply()
    }
}

/** The model a request uses when the account has no explicit default. */
fun defaultModelOf(context: Context, account: Account): String =
    account.model.ifBlank { ModelCache.models(context, account.id).firstOrNull().orEmpty() }

/** Stores the account's default model. */
fun setAccountModel(context: Context, accountId: String, model: String) {
    val updated = AccountStore.load(context).map { if (it.id == accountId) it.copy(model = model) else it }
    AccountStore.save(context, updated)
}

    }

    /** True when the error looks like a quota, credit or rate-limit failure. */
    fun isQuotaError(error: Throwable): Boolean {
        val message = error.message.orEmpty()
        return message.contains("HTTP 429") ||
            message.contains("HTTP 402") ||
            message.contains("credit", ignoreCase = true) ||
            message.contains("quota", ignoreCase = true)
    }

    /** Loads the model ids from the provider's /models endpoint. */
    suspend fun listModels(baseUrl: String, apiKey: String): Result<List<String>> = withContext(Dispatchers.IO) {
        try {
            val builder = Request.Builder()
                .url(baseUrl.trimEnd('/') + "/models")
                .get()
                .header("Accept", "application/json")
            if (apiKey.isNotEmpty()) builder.header("Authorization", "Bearer $apiKey")
            http.newCall(builder.build()).execute().use { response ->
                if (!response.isSuccessful) error("HTTP ${response.code}")
                val body = response.body?.string().orEmpty()
                val data = JSONObject(body).optJSONArray("data") ?: JSONArray()
                val ids = (0 until data.length()).mapNotNull { index ->
                    data.optJSONObject(index)?.optString("id")?.takeIf { it.isNotBlank() }
                }
                Result.success(ids.distinct().sorted())
            }
        } catch (e: CancellationException) {
            throw e
        } catch (e: Exception) {
            Result.failure(e)
        }
    }

    suspend fun stream(
        baseUrl: String,
        apiKey: String,
        model: String,
        messages: JSONArray,
        onDelta: (String) -> Unit
    ): Result<Unit> = withContext(Dispatchers.IO) {
        try {
            val payload = JSONObject()
                .put("model", model)
                .put("messages", messages)
                .put("stream", true)
            val builder = Request.Builder()
                .url(baseUrl.trimEnd('/') + "/chat/completions")
                .post(payload.toString().toRequestBody(jsonType))
                .header("Accept", "text/event-stream")
            if (apiKey.isNotEmpty()) builder.header("Authorization", "Bearer $apiKey")

            http.newCall(builder.build()).execute().use { response ->
                if (!response.isSuccessful) {
                    val detail = response.body?.string()?.take(300).orEmpty()
                    error("HTTP ${response.code} $detail")
                }
                val source = response.body?.source() ?: error("Empty response body")
                while (true) {
                    val line = source.readUtf8Line() ?: break
                    if (!line.startsWith("data:")) continue
                    val data = line.removePrefix("data:").trim()
                    if (data == "[DONE]") break
                    val delta = JSONObject(data)
                        .optJSONArray("choices")
                        ?.optJSONObject(0)
                        ?.optJSONObject("delta")
                        ?.optString("content", "")
                    if (!delta.isNullOrEmpty()) onDelta(delta)
                }
            }
            Result.success(Unit)
        } catch (e: CancellationException) {
            throw e
        } catch (e: Exception) {
            Result.failure(e)
        }
    }

package com.hiyako.app.security

import android.content.Context
import android.security.keystore.KeyGenParameterSpec
import android.security.keystore.KeyProperties
import android.util.Base64
import java.security.KeyStore
import javax.crypto.Cipher
import javax.crypto.KeyGenerator
import javax.crypto.SecretKey
import javax.crypto.spec.GCMParameterSpec

/**
 * Encrypts secrets (API keys) with an AES-256-GCM key that lives in the Android Keystore.
 * The plaintext key never touches disk and is never logged.
 * Stored format: base64(iv) + ":" + base64(ciphertext).
 */
object KeyVault {
    private const val KEY_ALIAS = "hiyako_key_vault_v1"
    private const val PREFS_NAME = "hiyako_secure_prefs"
    private const val TRANSFORMATION = "AES/GCM/NoPadding"
    private const val GCM_TAG_BITS = 128

    fun prefs(context: Context) = context.getSharedPreferences(PREFS_NAME, Context.MODE_PRIVATE)

    private fun secretKey(): SecretKey {
        val keyStore = KeyStore.getInstance("AndroidKeyStore").apply { load(null) }
        val existing = keyStore.getEntry(KEY_ALIAS, null) as? KeyStore.SecretKeyEntry
        if (existing != null) return existing.secretKey

        val generator = KeyGenerator.getInstance(KeyProperties.KEY_ALGORITHM_AES, "AndroidKeyStore")
        val spec = KeyGenParameterSpec.Builder(
            KEY_ALIAS,
            KeyProperties.PURPOSE_ENCRYPT or KeyProperties.PURPOSE_DECRYPT
        )
            .setBlockModes(KeyProperties.BLOCK_MODE_GCM)
            .setEncryptionPaddings(KeyProperties.ENCRYPTION_PADDING_NONE)
            .setKeySize(256)
            .build()
        generator.init(spec)
        return generator.generateKey()
    }

    fun encrypt(plain: String): String {
        val cipher = Cipher.getInstance(TRANSFORMATION)
        cipher.init(Cipher.ENCRYPT_MODE, secretKey())
        val cipherText = cipher.doFinal(plain.toByteArray(Charsets.UTF_8))
        val iv = Base64.encodeToString(cipher.iv, Base64.NO_WRAP)
        val data = Base64.encodeToString(cipherText, Base64.NO_WRAP)
        return "$iv:$data"
    }

    /** Returns null when the blob cannot be decrypted (for example after the keystore was reset). */
    fun decrypt(blob: String): String? {
        return try {
            val parts = blob.split(":", limit = 2)
            if (parts.size != 2) return null
            val iv = Base64.decode(parts[0], Base64.NO_WRAP)
            val data = Base64.decode(parts[1], Base64.NO_WRAP)
            val cipher = Cipher.getInstance(TRANSFORMATION)
            cipher.init(Cipher.DECRYPT_MODE, secretKey(), GCMParameterSpec(GCM_TAG_BITS, iv))
            String(cipher.doFinal(data), Charsets.UTF_8)
        } catch (e: Exception) {
            null
        }
    }
}

import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import okhttp3.MediaType.Companion.toMediaType

import okhttp3.Request
import okhttp3.RequestBody.Companion.toRequestBody
import org.json.JSONArray
import org.json.JSONObject
import java.util.Base64

data class Repo(val name: String, val fullName: String, val defaultBranch: String)

object GitHubApi {
    private val client = OkHttpClient.Builder().build()
    private val jsonMediaType = "application/json; charset=utf-8".toMediaType()

    private fun authRequest(url: String, token: String) = Request.Builder()
        .url(url)
        .header("Authorization", "Bearer $token")
        .header("Accept", "application/vnd.github.v3+json")
        .build()

    suspend fun getUserRepos(token: String): List<Repo> = withContext(Dispatchers.IO) {
        try {
            client.newCall(authRequest("https://api.github.com/user/repos?per_page=100&sort=updated", token)).execute().use { response ->
                if (!response.isSuccessful) return@withContext emptyList()
                val jsonArray = JSONArray(response.body?.string() ?: "[]")
                val repos = mutableListOf<Repo>()
                for (i in 0 until jsonArray.length()) {
                    val repo = jsonArray.getJSONObject(i)
                    repos.add(Repo(repo.getString("name"), repo.getString("full_name"), repo.getString("default_branch")))
                }
                return@withContext repos
            }
        } catch (e: Exception) { return@withContext emptyList() }
    }

    suspend fun getRepoFiles(token: String, owner: String, repo: String, branch: String): List<String> = withContext(Dispatchers.IO) {
        val files = mutableListOf<String>()
        try {
            val url = "https://api.github.com/repos/$owner/$repo/git/trees/$branch?recursive=1"
            client.newCall(authRequest(url, token)).execute().use { response ->
                if (!response.isSuccessful) return@withContext emptyList()
                val tree = JSONObject(response.body?.string() ?: "{}").optJSONArray("tree") ?: return@withContext emptyList()
                for (i in 0 until tree.length()) {
                    val item = tree.getJSONObject(i)
                    if (item.getString("type") == "blob") {
                        val path = item.getString("path")
                        if (!path.contains("build/") && !path.contains(".git/") && 
                            (path.endsWith(".kt") || path.endsWith(".xml") || path.endsWith(".kts") || 
                             path.endsWith(".gradle") || path.endsWith(".md") || path.endsWith(".json") || 
                             path.endsWith(".properties") || path.endsWith(".pro") || path.endsWith(".toml"))) {
                            files.add(path)
                        }
                    }
                }
            }
        } catch (e: Exception) { e.printStackTrace() }
        return@withContext files
    }

    suspend fun getFileContent(token: String, owner: String, repo: String, path: String): String? = withContext(Dispatchers.IO) {
        try {
            val url = "https://api.github.com/repos/$owner/$repo/contents/$path"
            client.newCall(authRequest(url, token)).execute().use { response ->
                if (!response.isSuccessful) return@withContext null
                val contentBase64 = JSONObject(response.body?.string() ?: "{}").optString("content").replace("\n", "")
                if (contentBase64.isBlank()) return@withContext null
                String(Base64.getDecoder().decode(contentBase64), Charsets.UTF_8)
            }
        } catch (e: Exception) { return@withContext null }
    }

    private suspend fun getFileSha(token: String, owner: String, repo: String, path: String): String? = withContext(Dispatchers.IO) {
        try {
            val url = "https://api.github.com/repos/$owner/$repo/contents/$path"
            client.newCall(authRequest(url, token)).execute().use { response ->
                if (response.isSuccessful) return@withContext JSONObject(response.body?.string() ?: "{}").optString("sha")
            }
        } catch (e: Exception) {}
        return@withContext null
    }

    suspend fun updateFile(token: String, owner: String, repo: String, path: String, newContent: String, commitMessage: String): Boolean = withContext(Dispatchers.IO) {
        try {
            val sha = getFileSha(token, owner, repo, path)
            val payload = JSONObject().apply {
                put("message", commitMessage)
                put("content", Base64.getEncoder().encodeToString(newContent.toByteArray(Charsets.UTF_8)))
                if (sha != null) put("sha", sha)
            }
            
            val request = Request.Builder()
                .url("https://api.github.com/repos/$owner/$repo/contents/$path")
                .header("Authorization", "Bearer $token")
                .header("Accept", "application/vnd.github.v3+json")
                .put(payload.toString().toRequestBody(jsonMediaType))
                .build()

            client.newCall(request).execute().use { response -> return@withContext response.isSuccessful }
        } catch (e: Exception) { return@withContext false }
    }

    suspend fun deleteFile(token: String, owner: String, repo: String, path: String, commitMessage: String): Boolean = withContext(Dispatchers.IO) {
        try {
            val sha = getFileSha(token, owner, repo, path) ?: return@withContext false
            val payload = JSONObject().apply {
                put("message", commitMessage)
                put("sha", sha)
            }
            
            val request = Request.Builder()
                .url("https://api.github.com/repos/$owner/$repo/contents/$path")
                .header("Authorization", "Bearer $token")
                .header("Accept", "application/vnd.github.v3+json")
                .delete(payload.toString().toRequestBody(jsonMediaType))
                .build()

            client.newCall(request).execute().use { response -> return@withContext response.isSuccessful }
        } catch (e: Exception) { return@withContext false }
    }

package com.hiyako.app.util

import android.content.Context
import android.net.Uri
import android.provider.OpenableColumns

fun queryDisplayName(context: Context, uri: Uri): String {
    var name: String? = null
    try {
        context.contentResolver.query(uri, arrayOf(OpenableColumns.DISPLAY_NAME), null, null, null)?.use { cursor ->
            if (cursor.moveToFirst()) {
                val index = cursor.getColumnIndex(OpenableColumns.DISPLAY_NAME)
                if (index >= 0) name = cursor.getString(index)
            }
        }
    } catch (e: Exception) {
        // ignore and fall back below
    }
    return name ?: uri.lastPathSegment ?: "file"
}

fun isRtlText(text: String): Boolean {
    for (ch in text) {
        if (Character.isLetter(ch)) {
            val direction = Character.getDirectionality(ch)
            return direction == Character.DIRECTIONALITY_RIGHT_TO_LEFT ||
                direction == Character.DIRECTIONALITY_RIGHT_TO_LEFT_ARABIC
        }
    }
    return false
}
