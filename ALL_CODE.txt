package com.hiyako.app

import android.app.NotificationChannel
import android.app.NotificationManager
import android.app.Service
import android.content.Intent
import android.os.Build
import android.os.IBinder
import android.os.PowerManager
import androidx.core.app.NotificationCompat

class NexusService : Service() {
    private var wakeLock: PowerManager.WakeLock? = null

    override fun onBind(intent: Intent?): IBinder? = null

    override fun onCreate() {
        super.onCreate()
        val powerManager = getSystemService(POWER_SERVICE) as PowerManager
        wakeLock = powerManager.newWakeLock(PowerManager.PARTIAL_WAKE_LOCK, "NexusAI::ProcessingWakeLock")
        wakeLock?.acquire(10 * 60 * 1000L) // 10 minutes max
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        val channelId = "nexus_fg_channel"
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val channel = NotificationChannel(channelId, "Nexus AI Processing", NotificationManager.IMPORTANCE_LOW)
            getSystemService(NotificationManager::class.java).createNotificationChannel(channel)
        }

        val notification = NotificationCompat.Builder(this, channelId)
            .setContentTitle("Nexus AI is working...")
            .setContentText("Generating response in the background")
            .setSmallIcon(android.R.drawable.ic_dialog_info)
            .setOngoing(true)
            .build()

        startForeground(1, notification)
        return START_NOT_STICKY
    }

    override fun onDestroy() {
        wakeLock?.let { if (it.isHeld) it.release() }
        super.onDestroy()
    }
}

@Composable
fun AnimatedLoadingText(type: String) {
    var dotCount by remember { mutableIntStateOf(1) }
    LaunchedEffect(Unit) {
        while (true) {
            delay(500)
            dotCount = if (dotCount == 3) 1 else dotCount + 1
        }
    }
    val baseText = when (type) {
        "IMAGE" -> "Creating image"
        "VIDEO" -> "Generating video"
        else -> "Thinking"
    }
    Text(text = "$baseText${".".repeat(dotCount)}", color = NexusCyan, fontSize = 14.sp)
}

package com.hiyako.app.ui.screens

import android.content.Context
import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.hiyako.app.data.GitHubApi
import com.hiyako.app.data.Repo
import com.hiyako.app.data.WorkspaceManager
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch

class WorkspaceViewModel : ViewModel() {
    private val _repos = MutableStateFlow<List<Repo>>(emptyList())
    val repos = _repos.asStateFlow()

    private val _isLoading = MutableStateFlow(false)
    val isLoading = _isLoading.asStateFlow()
    
    private val _loadingMessage = MutableStateFlow("")
    val loadingMessage = _loadingMessage.asStateFlow()

    private val _token = MutableStateFlow("")
    val token = _token.asStateFlow()

    fun loadToken(context: Context) {
        val prefs = context.getSharedPreferences("nexus_prefs", Context.MODE_PRIVATE)
        _token.value = prefs.getString("github_token", "") ?: ""
        if (_token.value.isNotBlank()) fetchRepos()
    }

    fun saveToken(context: Context, token: String) {
        val prefs = context.getSharedPreferences("nexus_prefs", Context.MODE_PRIVATE)
        prefs.edit().putString("github_token", token).apply()
        _token.value = token
        fetchRepos()
    }

    fun fetchRepos() {
        if (_token.value.isBlank()) return
        _isLoading.value = true
        viewModelScope.launch {
            _repos.value = GitHubApi.getUserRepos(_token.value)
            _isLoading.value = false
        }
    }

    fun selectRepoAndLoadFiles(repo: Repo, onComplete: () -> Unit) {
        _isLoading.value = true
        _loadingMessage.value = "Fetching file tree..."
        viewModelScope.launch {
            val owner = repo.fullName.split("/")[0]
            val filesList = GitHubApi.getRepoFiles(_token.value, owner, repo.name, repo.defaultBranch)
            val loadedFiles = mutableMapOf<String, String>()
            
            // تم إزالة الحد الأقصى لقراءة كل الملفات
            filesList.forEachIndexed { index, path -> 
                _loadingMessage.value = "Loading files: ${index + 1}/${filesList.size}..."
                val content = GitHubApi.getFileContent(_token.value, owner, repo.name, path)
                if (content != null) {
                    loadedFiles[path] = content
                }
            }
            
            WorkspaceManager.activeRepo = repo.fullName
            WorkspaceManager.projectFiles = loadedFiles
            WorkspaceManager.isLoaded = loadedFiles.isNotEmpty()
            
            _isLoading.value = false
            _loadingMessage.value = ""
            onComplete()
        }
    }
}

@Composable
fun UpdaterScreen(onBack: () -> Unit) {
    val context = LocalContext.current
    val scope = rememberCoroutineScope()
    var isDownloading by remember { mutableStateOf(false) }
    var statusText by remember { mutableStateOf("Click below to fetch the latest APK from GitHub.") }
    
    // يجب استبدال هذا برابط الـ Artifact المباشر من GitHub Actions
    // للأسف GitHub لا يوفر رابطاً مباشراً بدون مصادقة، لذا سنستخدم
    // رابط الـ Release (إن وجد) أو نطلب من المستخدم وضعه.
    val apkUrl = "https://github.com/alsbtv/Hiyako-/releases/latest/download/app-debug.apk" 

    Scaffold(
        containerColor = OledBlack,
        topBar = {
            TopAppBar(
                title = { Text("App Updater", color = TextPrimary, fontWeight = FontWeight.Bold) },
                navigationIcon = { IconButton(onClick = onBack) { Icon(Icons.Default.ArrowBack, null, tint = TextPrimary) } },
                colors = TopAppBarDefaults.topAppBarColors(containerColor = OledBlack)
            )
        }
    ) { padding ->
        Column(
            modifier = Modifier.padding(padding).fillMaxSize().padding(24.dp),
            horizontalAlignment = Alignment.CenterHorizontally,
            verticalArrangement = Arrangement.Center
        ) {
            Icon(Icons.Default.CloudDownload, null, tint = AccentBlue, modifier = Modifier.size(80.dp))
            Spacer(Modifier.height(24.dp))
            Text("Nexus Auto-Updater", color = TextPrimary, fontSize = 24.sp, fontWeight = FontWeight.Bold)
            Spacer(Modifier.height(8.dp))
            Text(statusText, color = TextSecondary, fontSize = 14.sp)
            Spacer(Modifier.height(32.dp))

            Button(
                onClick = {
                    isDownloading = true
                    statusText = "Downloading update..."
                    scope.launch(Dispatchers.IO) {
                        try {
                            val client = OkHttpClient()
                            val request = Request.Builder().url(apkUrl).build()
                            val response = client.newCall(request).execute()
                            
                            if (response.isSuccessful) {
                                val body = response.body
                                val file = File(context.cacheDir, "update.apk")
                                val fos = FileOutputStream(file)
                                body?.byteStream()?.copyTo(fos)
                                fos.close()
                                
                                withContext(Dispatchers.Main) {
                                    statusText = "Download complete. Installing..."
                                    val intent = Intent(Intent.ACTION_VIEW)
                                    val uri = androidx.core.content.FileProvider.getUriForFile(context, "${context.packageName}.provider", file)
                                    intent.setDataAndType(uri, "application/vnd.android.package-archive")
                                    intent.addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION)
                                    context.startActivity(intent)
                                    isDownloading = false
                                }
                            } else {
                                withContext(Dispatchers.Main) {
                                    statusText = "Error: Could not download APK. Check URL."
                                    isDownloading = false
                                }
                            }
                        } catch (e: Exception) {
                            withContext(Dispatchers.Main) {
                                statusText = "Error: ${e.message}"
                                isDownloading = false
                            }
                        }
                    }
                },
                enabled = !isDownloading,
                modifier = Modifier.fillMaxWidth().height(50.dp),
                colors = ButtonDefaults.buttonColors(containerColor = AccentBlue),
                shape = RoundedCornerShape(12.dp)
            ) {
                if (isDownloading) {
                    CircularProgressIndicator(color = Color.White, modifier = Modifier.size(24.dp))
                } else {
                    Text("Download & Install Update", color = Color.White, fontWeight = FontWeight.Bold)
                }
            }
        }
    }
}

data class AiModel(val name: String, val id: String)

class AiViewModel(application: Application) : AndroidViewModel(application) {
    private val db = AppDatabase.getDatabase(application)
    private val chatDao = db.chatDao()

    private val _messages = MutableStateFlow<List<Message>>(emptyList())
    val messages = _messages.asStateFlow()
    private val _isReady = MutableStateFlow(false)
    
    private val _isBuilding = MutableStateFlow(false)
    val isBuilding = _isBuilding.asStateFlow()
    
    private val _chatTitle = MutableStateFlow("New Chat")
    val chatTitle = _chatTitle.asStateFlow()

    private val _selectedModel = MutableStateFlow(AiModel("Default (Auto)", ""))
    val selectedModel = _selectedModel.asStateFlow()

    private val _availableModels = MutableStateFlow<List<AiModel>>(listOf(AiModel("Loading models...", "")))
    val availableModels = _availableModels.asStateFlow()
    private val _filteredModels = MutableStateFlow<List<AiModel>>(emptyList())
    val filteredModels = _filteredModels.asStateFlow()

    // لضمان عدم تعليق "Thinking..." للأبد
    private var currentRequestId: Long = 0

    init {
        viewModelScope.launch {
            try {
                chatDao.getAllMessages().collect { entities ->
                    _messages.value = entities.map { entity ->
                        val isThinking = entity.role == "ai" && entity.content == "Thinking..."
                        Message(entity.content, entity.role == "user", isLoading = isThinking, type = entity.messageType)
                    }
                }
            } catch (e: Exception) {
                chatDao.clearChat()
            }
        }
    }

    fun clearChat() {
        viewModelScope.launch {
            chatDao.clearChat()
            _chatTitle.value = "New Chat"
        }
    }

    fun setReady() { _isReady.value = true }
    fun selectModel(model: AiModel) { _selectedModel.value = model }
    
    fun onModelsReceived(jsonString: String) {
        try {
            val jsonArray = org.json.JSONArray(jsonString)
            val modelsList = mutableListOf<AiModel>()
            modelsList.add(AiModel("Default (Auto)", ""))
            for (i in 0 until jsonArray.length()) {
                val id = jsonArray.getJSONObject(i).optString("id")
                if (id.isNotBlank()) modelsList.add(AiModel(jsonArray.getJSONObject(i).optString("name", id), id))
            }
            _availableModels.value = modelsList.sortedBy { it.name }
            _filteredModels.value = _availableModels.value
        } catch (e: Exception) { e.printStackTrace() }
    }

    fun filterModels(query: String) {
        _filteredModels.value = if (query.isBlank()) _availableModels.value else _availableModels.value.filter { it.name.contains(query, ignoreCase = true) }
    }

    fun sendMessage(text: String, webView: android.webkit.WebView?, forceType: String? = null) {
        if (text.isBlank() || !_isReady.value) return
        
        // تصنيف بسيط وسريع (بدون الاعتماد على نموذج AI منفصل)
        val msgType = forceType ?: when {
            text.startsWith("ارسم", ignoreCase = true) -> "IMAGE"
            text.startsWith("سويلي صورة", ignoreCase = true) -> "IMAGE"
            text.startsWith("اصنع صورة", ignoreCase = true) -> "IMAGE"
            text.startsWith("generate an image", ignoreCase = true) -> "IMAGE"
            text.startsWith("create an image", ignoreCase = true) -> "IMAGE"
            text.startsWith("create a video", ignoreCase = true) -> "VIDEO"
            text.startsWith("generate a video", ignoreCase = true) -> "VIDEO"
            else -> "TEXT"
        }

        executeSendMessage(text, webView, msgType)
    }

    private fun executeSendMessage(text: String, webView: android.webkit.WebView?, type: String) {
        if (_messages.value.isEmpty() || _messages.value.size == 1) {
            _chatTitle.value = text.take(25) + if (text.length > 25) "..." else ""
        }

        val intent = Intent(getApplication(), NexusService::class.java)
        ContextCompat.startForegroundService(getApplication(), intent)

        val requestId = ++currentRequestId

        viewModelScope.launch {
            chatDao.insertMessage(ChatEntity(role = "user", content = text))
            chatDao.insertMessage(ChatEntity(role = "ai", content = "Thinking...", messageType = type)) 
        }

        val js: String = when (type) {
            "IMAGE" -> "window.generateImage(\"${text.replace("\"", "\\\"")}\");"
            "VIDEO" -> "window.generateVideo(\"${text.replace("\"", "\\\"")}\");"
            else -> {
                var fullPrompt = ""
                if (WorkspaceManager.isLoaded) {
                    val mentionedFiles = WorkspaceManager.projectFiles.filterKeys { path ->
                        text.contains(path.substringAfterLast("/")) || text.contains(path)
                    }
                    if (mentionedFiles.isNotEmpty()) {
                        fullPrompt += "Here are the files:\n"
                        mentionedFiles.forEach { (path, content) -> fullPrompt += "--- $path ---\n$content\n" }
                    } else {
                        fullPrompt += "Project files:\n" + WorkspaceManager.projectFiles.keys.joinToString("\n")
                    }
                    fullPrompt += "\nUser Request: $text\nOutput FULL code with [FILE: path] format if needed."
                } else {
                    fullPrompt = text
                }
                "window.askPuter(\"${fullPrompt.replace("\"", "\\\"").replace("\n", "\\n")}\", \"${_selectedModel.value.id}\");"
            }
        }
        webView?.evaluateJavascript(js, null)
    }


    fun stopGeneration() {
        currentRequestId++
        getApplication<Application>().stopService(Intent(getApplication(), NexusService::class.java))
        viewModelScope.launch {
            chatDao.deleteThinkingMessage()
            chatDao.insertMessage(ChatEntity(role = "ai", content = "⏹️ Generation stopped by user.", messageType = "TEXT"))
        }
    }

    fun applyChanges(token: String, aiResponse: String, context: Context) {
        if (WorkspaceManager.activeRepo.isBlank()) return
        _isBuilding.value = true
        val parts = WorkspaceManager.activeRepo.split("/")
        val owner = parts[0]
        val repo = parts[1]

        val intent = Intent(context, NexusService::class.java)
        ContextCompat.startForegroundService(context, intent)

        val filePattern = Pattern.compile("\\[FILE:\\s*(.*?)\\]\\s*```[a-zA-Z]*\\n([\\s\\S]*?)```")
        val fileMatcher = filePattern.matcher(aiResponse)
        val filesToUpdate = mutableMapOf<String, String>()
        while (fileMatcher.find()) {
            val path = fileMatcher.group(1)?.trim() ?: continue
            val code = fileMatcher.group(2)?.trim() ?: continue
            filesToUpdate[path] = code
        }

        val deletePattern = Pattern.compile("\\[DELETE:\\s*(.*?)\\]")
        val deleteMatcher = deletePattern.matcher(aiResponse)
        val filesToDelete = mutableListOf<String>()
        while (deleteMatcher.find()) {
            deleteMatcher.group(1)?.trim()?.let { filesToDelete.add(it) }
        }

        if (filesToUpdate.isEmpty() && filesToDelete.isEmpty()) {
            _isBuilding.value = false
            context.stopService(Intent(context, NexusService::class.java))
            return
        }

        viewModelScope.launch {
            var successCount = 0
            filesToUpdate.forEach { (path, code) ->
                if (GitHubApi.updateFile(token, owner, repo, path, code, "AI Auto-Update: Modified $path")) successCount++
            }
            filesToDelete.forEach { path ->
                if (GitHubApi.deleteFile(token, owner, repo, path, "AI Auto-Update: Deleted $path")) successCount++
            }
            
            _isBuilding.value = false
            context.stopService(Intent(context, NexusService::class.java))
            val successMsg = "✅ Successfully processed $successCount files on GitHub."
            chatDao.insertMessage(ChatEntity(role = "ai", content = successMsg, messageType = "TEXT"))
            sendNotification("Build Started", "Pushed changes. Building...")
        }
    }

    private fun sendNotification(title: String, message: String) {
        val context = getApplication<Application>()
        val notificationManager = context.getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
        val channelId = "nexus_channel"

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val channel = NotificationChannel(channelId, "Nexus AI", NotificationManager.IMPORTANCE_DEFAULT)
            notificationManager.createNotificationChannel(channel)
        }

        val notification = NotificationCompat.Builder(context, channelId)
            .setSmallIcon(android.R.drawable.ic_dialog_info)
            .setContentTitle(title)
            .setContentText(message)
            .setPriority(NotificationCompat.PRIORITY_DEFAULT)
            .build()

        notificationManager.notify(2, notification)
    }

package com.hiyako.app.data

import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import okhttp3.MediaType.Companion.toMediaType
import okhttp3.OkHttpClient
import okhttp3.Request
import okhttp3.RequestBody.Companion.toRequestBody
import org.json.JSONArray
import org.json.JSONObject
import java.util.Base64

data class Repo(val name: String, val fullName: String, val defaultBranch: String)

object GitHubApi {
    private val client = OkHttpClient.Builder().build()
    private val jsonMediaType = "application/json; charset=utf-8".toMediaType()

    private fun authRequest(url: String, token: String) = Request.Builder()
        .url(url)
        .header("Authorization", "Bearer $token")
        .header("Accept", "application/vnd.github.v3+json")
        .build()

    suspend fun getUserRepos(token: String): List<Repo> = withContext(Dispatchers.IO) {
        try {
            client.newCall(authRequest("https://api.github.com/user/repos", token)).execute().use { response ->
                if (!response.isSuccessful) return@withContext emptyList()
                val jsonArray = JSONArray(response.body?.string() ?: "[]")
                val repos = mutableListOf<Repo>()
                for (i in 0 until jsonArray.length()) {
                    val repo = jsonArray.getJSONObject(i)
                    repos.add(Repo(repo.getString("name"), repo.getString("full_name"), repo.getString("default_branch")))
                }
                return@withContext repos
            }
        } catch (e: Exception) { return@withContext emptyList() }
    }

    suspend fun getRepoFiles(token: String, owner: String, repo: String, branch: String): List<String> = withContext(Dispatchers.IO) {
        val files = mutableListOf<String>()
        try {
            val url = "https://api.github.com/repos/$owner/$repo/git/trees/$branch?recursive=1"
            client.newCall(authRequest(url, token)).execute().use { response ->
                if (!response.isSuccessful) return@withContext emptyList()
                val tree = JSONObject(response.body?.string() ?: "{}").optJSONArray("tree") ?: return@withContext emptyList()
                for (i in 0 until tree.length()) {
                    val item = tree.getJSONObject(i)
                    if (item.getString("type") == "blob") {
                        val path = item.getString("path")
                        if (!path.contains("build/") && !path.contains(".git/") && (path.endsWith(".kt") || path.endsWith(".xml") || path.endsWith(".kts") || path.endsWith(".gradle"))) {
                            files.add(path)
                        }
                    }
                }
            }
        } catch (e: Exception) { e.printStackTrace() }
        return@withContext files
    }

    suspend fun getFileContent(token: String, owner: String, repo: String, path: String): String? = withContext(Dispatchers.IO) {
        try {
            val url = "https://api.github.com/repos/$owner/$repo/contents/$path"
            client.newCall(authRequest(url, token)).execute().use { response ->
                if (!response.isSuccessful) return@withContext null
                val contentBase64 = JSONObject(response.body?.string() ?: "{}").optString("content").replace("\n", "")
                if (contentBase64.isBlank()) return@withContext null
                String(Base64.getDecoder().decode(contentBase64))
            }
        } catch (e: Exception) { return@withContext null }
    }

    private suspend fun getFileSha(token: String, owner: String, repo: String, path: String): String? = withContext(Dispatchers.IO) {
        try {
            val url = "https://api.github.com/repos/$owner/$repo/contents/$path"
            client.newCall(authRequest(url, token)).execute().use { response ->
                if (response.isSuccessful) {
                    return@withContext JSONObject(response.body?.string() ?: "{}").optString("sha")
                }
            }
        } catch (e: Exception) {}
        return@withContext null
    }

    suspend fun updateFile(token: String, owner: String, repo: String, path: String, newContent: String, commitMessage: String): Boolean = withContext(Dispatchers.IO) {
        try {
            val sha = getFileSha(token, owner, repo, path)
            val payload = JSONObject().apply {
                put("message", commitMessage)
                put("content", Base64.getEncoder().encodeToString(newContent.toByteArray()))
                if (sha != null) put("sha", sha)
            }
            
            val request = Request.Builder()
                .url("https://api.github.com/repos/$owner/$repo/contents/$path")
                .header("Authorization", "Bearer $token")
                .header("Accept", "application/vnd.github.v3+json")
                .put(payload.toString().toRequestBody(jsonMediaType))
                .build()

            client.newCall(request).execute().use { response -> return@withContext response.isSuccessful }
        } catch (e: Exception) { return@withContext false }
    }

    // دالة حذف الملفات
    suspend fun deleteFile(token: String, owner: String, repo: String, path: String, commitMessage: String): Boolean = withContext(Dispatchers.IO) {
        try {
            val sha = getFileSha(token, owner, repo, path) ?: return@withContext false
            val payload = JSONObject().apply {
                put("message", commitMessage)
                put("sha", sha)
            }
            
            val request = Request.Builder()
                .url("https://api.github.com/repos/$owner/$repo/contents/$path")
                .header("Authorization", "Bearer $token")
                .header("Accept", "application/vnd.github.v3+json")
                .delete(payload.toString().toRequestBody(jsonMediaType))
                .build()

            client.newCall(request).execute().use { response -> return@withContext response.isSuccessful }
        } catch (e: Exception) { return@withContext false }
    }
}

                val instance = Room.databaseBuilder(
                    context.applicationContext,
                    AppDatabase::class.java,
                    "nexus_database"
                ).fallbackToDestructiveMigration().build()
                INSTANCE = instance
                instance
            }
        }
    }

@Composable
fun ChatRowItem(

package com.nexus.ai.ui.theme

import android.app.Activity
import androidx.compose.foundation.isSystemInDarkTheme
import androidx.compose.material3.MaterialTheme
import androidx.compose.material3.darkColorScheme
import androidx.compose.material3.lightColorScheme
import androidx.compose.runtime.Composable
import androidx.compose.runtime.SideEffect
import androidx.compose.ui.platform.LocalView
import androidx.core.view.WindowCompat

private val LightColorScheme = lightColorScheme(
    primary = LightAccentBlue,
    background = LightBackground,
    surface = LightSurface,
    onPrimary = LightBackground,
    onBackground = LightTextPrimary,
    onSurface = LightTextPrimary,
    surfaceVariant = LightUserBubble,
    onSurfaceVariant = LightTextSecondary,
    outline = LightDivider
)

private val DarkColorScheme = darkColorScheme(
    primary = DarkAccentBlue,
    background = DarkBackground,
    surface = DarkSurface,
    onPrimary = DarkBackground,
    onBackground = DarkTextPrimary,
    onSurface = DarkTextPrimary,
    surfaceVariant = DarkUserBubble,
    onSurfaceVariant = DarkTextSecondary,
    outline = DarkDivider
)

@Composable
fun NexusTheme(
    darkTheme: Boolean = isSystemInDarkTheme(),
    content: @Composable () -> Unit
) {
    val colorScheme = if (darkTheme) DarkColorScheme else LightColorScheme

    val view = LocalView.current
    if (!view.isInEditMode) {
        SideEffect {
            val window = (view.context as Activity).window
            window.statusBarColor = android.graphics.Color.TRANSPARENT
            WindowCompat.getInsetsController(window, view).isAppearanceLightStatusBars = !darkTheme
        }
    }

    MaterialTheme(
        colorScheme = colorScheme,
        content = content
    )
}

import androidx.compose.runtime.mutableStateOf

data class ChatItem(
    val id: String,
    val title: String,
    val timestamp: Long,
    val isPinned: Boolean = false
)

class DrawerViewModel : ViewModel() {
    // قائمة المحادثات تبدأ فارغة (لا بيانات وهمية)
    val chatList = mutableStateListOf<ChatItem>()
    
    var searchQuery by mutableStateOf("")
    var isMultiSelectMode by mutableStateOf(false)
    val selectedChats = mutableStateListOf<String>()

    fun addChat(title: String) {
        val newChat = ChatItem(
            id = System.currentTimeMillis().toString(),
            title = title,
            timestamp = System.currentTimeMillis()
        )
        chatList.add(0, newChat) // إضافة المحادثة في الأعلى
    }

    fun toggleMultiSelect(chatId: String) {
        if (selectedChats.contains(chatId)) {
            selectedChats.remove(chatId)
        } else {
            selectedChats.add(chatId)
        }
        isMultiSelectMode = selectedChats.isNotEmpty()
    }

    fun exitMultiSelect() {
        selectedChats.clear()
        isMultiSelectMode = false
    }

    fun deleteSelected() {
        chatList.removeAll { selectedChats.contains(it.id) }
        exitMultiSelect()
    }

    fun pinSelected() {
        for (i in chatList.indices) {
            if (selectedChats.contains(chatList[i].id)) {
                chatList[i] = chatList[i].copy(isPinned = true)
            }
        }
        exitMultiSelect()
    }

    fun deleteSingleChat(chatId: String) {
        chatList.removeAll { it.id == chatId }
    }

    fun renameSingleChat(chatId: String, newTitle: String) {
        val index = chatList.indexOfFirst { it.id == chatId }
        if (index != -1) {
            chatList[index] = chatList[index].copy(title = newTitle)
        }
    }
}

package eu.kanade.tachiyomi.torrentutils

import aniyomi.core.common.torrent.DisabledTorrServerException
import aniyomi.core.common.torrent.TorrentHelpers
import aniyomi.core.common.torrent.TorrentServerApi
import aniyomi.core.common.torrent.model.Torrent
import eu.kanade.tachiyomi.network.NetworkHelper
import eu.kanade.tachiyomi.network.get
import eu.kanade.tachiyomi.torrentutils.model.DeadTorrentException
import eu.kanade.tachiyomi.torrentutils.model.TorrentFile
import eu.kanade.tachiyomi.torrentutils.model.TorrentInfo
import kotlinx.coroutines.runBlocking
import uy.kohesive.injekt.injectLazy
import java.net.SocketTimeoutException

object TorrentUtils {
    private val torrentServerApi: TorrentServerApi by injectLazy()
    private val network: NetworkHelper by injectLazy()

    suspend fun getTorrentInfo(
        url: String,
        title: String,
    ): TorrentInfo {
        val torrent: Torrent = if (url.startsWith("magnet")) {
            // Magnet links need to be added to the torrent server to retrieve their
            // information
            try {
                torrentServerApi.addTorrent(url, title, "", "", false)
            } catch (_: SocketTimeoutException) {
                throw DeadTorrentException()
            } catch (_: Exception) {
                throw DisabledTorrServerException()
            }
        } else {
            // For torrent files we can parse the information out of the file itself
            // without starting the torrent server
            network.client.get(url).use { response ->
                TorrentHelpers.parseTorrentDetailsFromTorrentFileContent(response.body.byteStream())
            }
        }
        return torrentToTorrentInfo(torrent, title)
    }

    // A suspend function has a different signature in the JVM than a regular function (an additional Continuation
    // parameter is added by the Kotlin compiler). We add another overload of getTorrentInfo that is not a suspend
    // function so that extensions targetting other forks where getTorrentInfo was not a suspend function can still
    // work.
    @Deprecated(
        message = "This overload of getTorrentInfo exists only for binary compatibility with extensions targeting" +
            " other forks where getTorrentInfo was not a suspend function",
        level = DeprecationLevel.HIDDEN,
    )
    @JvmName("getTorrentInfo")
    fun blockingShimForGetTorrentInfo(
        url: String,
        title: String,
    ): TorrentInfo {
        return runBlocking {
            getTorrentInfo(url, title)
        }
    }

    private fun torrentToTorrentInfo(torrent: Torrent, overrideTitle: String?): TorrentInfo {
        return TorrentInfo(
            overrideTitle ?: torrent.title,
            torrent.fileStats?.map { file ->
                TorrentFile(file.path, file.id ?: 0, file.length, torrent.hash!!, torrent.trackers ?: emptyList())
            } ?: emptyList(),
            torrent.hash!!,
            torrent.torrentSize ?: -1,
            torrent.trackers ?: emptyList(),
        )
    }
}

package eu.kanade.tachiyomi.animesource.model

sealed class AnimeFilter<T>(val name: String, var state: T) {
    open class Header(name: String) : AnimeFilter<Any>(name, 0)
    open class Separator(name: String = "") : AnimeFilter<Any>(name, 0)
    abstract class Select<V>(name: String, val values: Array<V>, state: Int = 0) : AnimeFilter<Int>(
        name,
        state,
    )
    abstract class Text(name: String, state: String = "") : AnimeFilter<String>(name, state)
    abstract class CheckBox(name: String, state: Boolean = false) : AnimeFilter<Boolean>(
        name,
        state,
    )
    abstract class TriState(name: String, state: Int = STATE_IGNORE) : AnimeFilter<Int>(name, state) {
        fun isIgnored() = state == STATE_IGNORE
        fun isIncluded() = state == STATE_INCLUDE
        fun isExcluded() = state == STATE_EXCLUDE

        companion object {
            const val STATE_IGNORE = 0
            const val STATE_INCLUDE = 1
            const val STATE_EXCLUDE = 2
        }
    }

    abstract class Group<V>(name: String, state: List<V>) : AnimeFilter<List<V>>(name, state)

    abstract class Sort(name: String, val values: Array<String>, state: Selection? = null) :
        AnimeFilter<Sort.Selection?>(name, state) {
        data class Selection(val index: Int, val ascending: Boolean)
    }

    override fun equals(other: Any?): Boolean {
        if (this === other) return true
        if (other !is AnimeFilter<*>) return false

        return name == other.name && state == other.state
    }

    override fun hashCode(): Int {
        var result = name.hashCode()
        result = 31 * result + (state?.hashCode() ?: 0)
        return result
    }
}

     * Returns the url of the given string without the scheme and domain.
     *
     * @param orig the full url.
     */
    private fun getUrlWithoutDomain(orig: String): String {
        return try {
            val uri = URI(orig)
            var out = uri.path
            if (uri.query != null) {
                out += "?" + uri.query
            }
            if (uri.fragment != null) {
                out += "#" + uri.fragment
            }
            out
        } catch (_: URISyntaxException) {
            orig
        }
    }

package eu.kanade.tachiyomi.source.model

sealed class Filter<T>(val name: String, var state: T) {
    open class Header(name: String) : Filter<Any>(name, 0)
    open class Separator(name: String = "") : Filter<Any>(name, 0)
    abstract class Select<V>(name: String, val values: Array<V>, state: Int = 0) : Filter<Int>(
        name,
        state,
    )
    abstract class Text(name: String, state: String = "") : Filter<String>(name, state)
    abstract class CheckBox(name: String, state: Boolean = false) : Filter<Boolean>(name, state)
    abstract class TriState(name: String, state: Int = STATE_IGNORE) : Filter<Int>(name, state) {
        fun isIgnored() = state == STATE_IGNORE
        fun isIncluded() = state == STATE_INCLUDE
        fun isExcluded() = state == STATE_EXCLUDE

        companion object {
            const val STATE_IGNORE = 0
            const val STATE_INCLUDE = 1
            const val STATE_EXCLUDE = 2
        }
    }

    abstract class Group<V>(name: String, state: List<V>) : Filter<List<V>>(name, state)

    abstract class Sort(name: String, val values: Array<String>, state: Selection? = null) :
        Filter<Sort.Selection?>(name, state) {
        data class Selection(val index: Int, val ascending: Boolean)
    }

    override fun equals(other: Any?): Boolean {
        if (this === other) return true
        if (other !is Filter<*>) return false

        return name == other.name && state == other.state
    }

    override fun hashCode(): Int {
        var result = name.hashCode()
        result = 31 * result + (state?.hashCode() ?: 0)
        return result
    }
}

import kotlinx.coroutines.flow.asStateFlow
import kotlinx.serialization.Serializable

@Serializable
open class Page(
    val index: Int,
    val url: String = "",
    var imageUrl: String? = null,
    @Transient var uri: Uri? = null, // Deprecated but can't be deleted due to extensions
) : ProgressListener {

    val number: Int
        get() = index + 1

    @Transient
    private val _statusFlow = MutableStateFlow(State.QUEUE)

    @Transient
    val statusFlow = _statusFlow.asStateFlow()
    var status: State
        get() = _statusFlow.value
        set(value) {
            _statusFlow.value = value
        }

    @Transient
    private val _progressFlow = MutableStateFlow(0)

    @Transient
    val progressFlow = _progressFlow.asStateFlow()
    var progress: Int
        get() = _progressFlow.value
        set(value) {
            _progressFlow.value = value
        }

    override fun update(bytesRead: Long, contentLength: Long, done: Boolean) {
        progress = if (contentLength > 0) {
            (100 * bytesRead / contentLength).toInt()
        } else {
            -1
        }
    }

    enum class State {
        QUEUE,
        LOAD_PAGE,
        DOWNLOAD_IMAGE,
        READY,
        ERROR,
    }
}

/**
 * A simple implementation for sources from a website.
 */
@Suppress("unused")
abstract class HttpSource : CatalogueSource {

    /**
     * Network service.
     */
    protected val network: NetworkHelper by injectLazy()

    /**
     * Base url of the website without the trailing slash, like: http://mysite.com
     */
    abstract val baseUrl: String

    /**
     * Version id used to generate the source id. If the site completely changes and urls are
     * incompatible, you may increase this value and it'll be considered as a new source.
     */
    open val versionId = 1

    /**
     * ID of the source. By default it uses a generated id using the first 16 characters (64 bits)
     * of the MD5 of the string `"${name.lowercase()}/$lang/$versionId"`.
     *
     * The ID is generated by the [generateId] function, which can be reused if needed
     * to generate outdated IDs for cases where the source name or language needs to
     * be changed but migrations can be avoided.
     *
     * Note: the generated ID sets the sign bit to `0`.
     */
    override val id by lazy { generateId(name, lang, versionId) }

    /**
     * Headers used for requests.
     */
    val headers: Headers by lazy { headersBuilder().build() }

    /**
     * Default network client for doing requests.
     */
    open val client: OkHttpClient
        get() = network.client

    /**
     * Generates a unique ID for the source based on the provided [name], [lang] and
     * [versionId]. It will use the first 16 characters (64 bits) of the MD5 of the string
     * `"${name.lowercase()}/$lang/$versionId"`.
     *
     * Note: the generated ID sets the sign bit to `0`.
     *
     * Can be used to generate outdated IDs, such as when the source name or language
     * needs to be changed but migrations can be avoided.
     *
     * @since extensions-lib 1.5
     * @param name [String] the name of the source
     * @param lang [String] the language of the source
     * @param versionId [Int] the version ID of the source
     * @return a unique ID for the source
     */
    @Suppress("MemberVisibilityCanBePrivate")
    protected fun generateId(name: String, lang: String, versionId: Int): Long {
        val key = "${name.lowercase()}/$lang/$versionId"
        val bytes = MessageDigest.getInstance("MD5").digest(key.toByteArray())
        return (0..7).map { bytes[it].toLong() and 0xff shl 8 * (7 - it) }.reduce(Long::or) and Long.MAX_VALUE
    }

fun Element.selectText(css: String, defaultValue: String? = null): String? {
    return select(css).first()?.text() ?: defaultValue
}

fun Element.selectInt(css: String, defaultValue: Int = 0): Int {
    return select(css).first()?.text()?.toInt() ?: defaultValue
}

fun Element.attrOrText(css: String): String {
    return if (css != "text") attr(css) else text()
}

/**
 * Returns a Jsoup document for this response.
 * @param html the body of the response. Use only if the body was read before calling this method.

    return Jsoup.parse(html ?: body.string(), request.url.toString())
